"use client";
import { Suspense, useState } from "react";
import { useSearchParams } from "next/navigation";
import { login, invalidateAuthStatusCache } from "@/lib/auth";
import { normalizeInternalReturnPath, inheritLoginHash } from "@/shared/auth/return-url";
import MurikahSocialButtons from "@/components/auth/MurikahSocialButtons";

function AccountForm({ signup }: { signup: boolean }) {
  const params = useSearchParams();
  signup = signup || params.get("signup") === "1";
  const next = normalizeInternalReturnPath(params.get("next"));
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [confirm, setConfirm] = useState("");
  const [error, setError] = useState(params.get("oauth_error") || "");
  const [busy, setBusy] = useState(false);
  async function submit(event: React.FormEvent) {
    event.preventDefault();
    setError("");
    if (signup && password !== confirm) { setError("The passwords do not match."); return; }
    setBusy(true);
    try {
      if (signup) {
        const res = await fetch("/api/murikah/access/signup", { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ username, password }) });
        if (!res.ok) {
          const body = await res.json().catch(() => ({}));
          throw new Error(typeof body.detail === "string" ? body.detail : "Check your account details and try again.");
        }
        invalidateAuthStatusCache();
      } else {
        const result = await login(username, password);
        if (!result.ok) throw new Error(result.error || "Sign in failed.");
      }
      // Full navigation resets cached guest identity and account-scoped providers.
      window.location.assign(inheritLoginHash(next, window.location.hash));
    } catch (e) { setError(e instanceof Error ? e.message : "Couldn't reach Tutor. Please try again."); setBusy(false); }
  }
  const inputClass = "w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3.5 py-2.5 text-sm focus:ring-2 focus:ring-[#A9822E]/30";
  return <div className="w-full max-w-sm px-5 py-10">
    <div className="mb-8 text-center"><div className="inline-flex rounded-lg bg-[#1E2A30] px-4 py-2 font-semibold text-white">Murikah <span className="mx-3 text-[#A9822E]">|</span> Tutor</div>
      <h1 className="mt-5 text-2xl font-semibold">{signup ? "Create your account" : "Welcome back"}</h1>
      <p className="mt-2 text-sm text-[var(--muted-foreground)]">{signup ? "Keep your guest work and continue learning." : "Sign in to continue your learning."}</p>
    </div>
    {error && <p role="alert" className="mb-4 rounded-lg bg-red-500/10 p-3 text-sm text-red-600">{error}</p>}
    <MurikahSocialButtons next={next} />
    <form onSubmit={submit} className="space-y-5 rounded-2xl border border-[var(--border)] p-6 shadow-sm">
      <div><label htmlFor="username" className="mb-2 block text-sm font-medium">Email or username</label><input id="username" autoComplete="username" maxLength={254} required value={username} onChange={e => setUsername(e.target.value)} className={inputClass} /></div>
      <div><label htmlFor="password" className="mb-2 block text-sm font-medium">Password</label><input id="password" type="password" autoComplete={signup ? "new-password" : "current-password"} minLength={signup ? 8 : undefined} required value={password} onChange={e => setPassword(e.target.value)} className={inputClass} />{signup && <p className="mt-1 text-xs text-[var(--muted-foreground)]">Use at least 8 characters.</p>}</div>
      {signup && <div><label htmlFor="confirm" className="mb-2 block text-sm font-medium">Confirm password</label><input id="confirm" type="password" autoComplete="new-password" required value={confirm} onChange={e => setConfirm(e.target.value)} className={inputClass} /></div>}
      <button type="submit" disabled={busy} className="w-full rounded-lg bg-[#1E2A30] px-4 py-3 font-semibold text-white disabled:opacity-50">{busy ? "Please wait…" : signup ? "Create account" : "Sign in"}</button>
    </form>
    <p className="mt-6 text-center text-sm">{signup ? "Already have an account? " : "New to Murikah Tutor? "}<a className="font-semibold underline" href={`${signup ? "/login" : "/register"}?next=${encodeURIComponent(next)}`}>{signup ? "Sign in" : "Sign up"}</a></p>
    <p className="mt-4 text-center text-sm"><a href="/">Continue as guest</a></p>
  </div>;
}
export default function MurikahAccountPage({ signup = false }: { signup?: boolean }) {
  return <Suspense fallback={<p>Loading…</p>}><AccountForm signup={signup} /></Suspense>;
}
