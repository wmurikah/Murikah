function formatDate(timestamp: number, withTime = false) {
  if (!timestamp) return "Not scheduled";
  const date = new Date(timestamp * 1000);
  return new Intl.DateTimeFormat(undefined, {
    day: "numeric",
    month: "short",
    year: date.getFullYear() !== new Date().getFullYear() ? "numeric" : undefined,
    hour: withTime ? "numeric" : undefined,
    minute: withTime ? "2-digit" : undefined,
  }).format(date);
}

function sentenceLabel(value: string) {
  const normalized = value.replace(/[_-]+/g, " ").trim();
  return normalized ? normalized[0].toUpperCase() + normalized.slice(1) : normalized;
}

function requestId(prefix: string) {
  const value =
    typeof crypto !== "undefined" && "randomUUID" in crypto
      ? crypto.randomUUID()
      : String(Date.now()) + "-" + Math.random().toString(16).slice(2);
  return prefix + "-" + value;
}

async function responseError(response: Response, fallback: string) {
  try {
    const payload = (await response.json()) as { detail?: string; message?: string };
    return payload.detail || payload.message || fallback;
  } catch {
    return fallback;
  }
}

async function readNdjson(response: Response, onEvent: (event: StreamEvent) => void) {
  if (!response.body) throw new Error("The response connection ended before any content arrived.");
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  for (;;) {
    const { value, done } = await reader.read();
    buffer += decoder.decode(value || new Uint8Array(), { stream: !done });
    const lines = buffer.split("\n");
    buffer = lines.pop() || "";
    for (const line of lines) {
      if (!line.trim()) continue;
      onEvent(JSON.parse(line) as StreamEvent);
    }
    if (done) break;
  }
  if (buffer.trim()) onEvent(JSON.parse(buffer) as StreamEvent);
}

    <article className="rounded-xl border border-[var(--border)] bg-[var(--card)] p-4">
      <div className="flex flex-wrap items-start justify-between gap-3">
        <div className="min-w-0">
          <p className="text-xs font-semibold uppercase tracking-[0.08em] text-[var(--muted-foreground)]">
            {sentenceLabel(task.category || "Work")}
          </p>
          <h3 className="mt-1 font-semibold leading-6">{task.title}</h3>
        </div>
        <span className="rounded-full border border-[var(--border)] px-2.5 py-1 text-xs font-medium">{sentenceLabel(task.work_status || task.status)}</span>
      </div>
      <div className="mt-3 flex flex-wrap gap-x-5 gap-y-1 text-xs text-[var(--muted-foreground)]">
        <span>{task.due_at ? "Due " + formatDate(task.due_at) : "No deadline set"}</span>
        <span>{person?.name ? "Assigned by " + person.name : "Assigned by workplace"}</span>
      </div>
      <button
        type="button"
        onClick={onOpen}
        className="mt-4 min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold hover:bg-[var(--background)] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#A9822E]"
      >
        View assignment
      </button>
    </article>
  );
}

function TaskDetail({
  task,
  people,
  internshipId,
  readOnly,
  onChanged,
  onClose,
}: {
  task: InternshipTaskSummary;
  people: InternshipPerson[];
  internshipId: string;
  readOnly: boolean;
  onChanged: () => Promise<void> | void;
  onClose: () => void;
}) {
  const person = people.find((item) => item.actor_id === task.assigned_by_actor_id);
  const [busy, setBusy] = useState("");
  const [actionError, setActionError] = useState("");
  const [drafts, setDrafts] = useState<Record<string, string>>({});
  const [editing, setEditing] = useState<Record<string, boolean>>({});

  const textDeliverables = new Set([
    "memo","report","working_paper","issue_summary","research_brief","policy_draft",
    "email","client_response","meeting_notes","requirements_document","test_plan",
    "risk_register","procurement_evaluation","project_plan","design_brief","incident_report",
  ]);
  const artifactFor = (deliverable: string) =>
    (task.work_artifacts || []).find((item) => item.deliverable_type === deliverable);

  async function callJson(path: string, payload: Record<string, unknown>) {
    const response = await fetch(API + path, {
      method: "POST",
      credentials: "include",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(payload),
    });
    if (!response.ok) throw new Error(await responseError(response, "That work update could not be completed. Try again."));
    return (await response.json()) as Record<string, any>;
  }

  async function acknowledge() {
    setBusy("ack"); setActionError("");
    try {
      await callJson("/tasks/" + encodeURIComponent(task.task_id) + "/acknowledge", {
        internship_id: internshipId,
        request_id: requestId("ack"),
      });
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "The assignment could not be acknowledged.");
    } finally { setBusy(""); }
  }

  async function createArtifact(deliverable: string) {
    const existing = artifactFor(deliverable);
    if (existing) return existing;
    const result = await callJson("/artifacts", {
      internship_id: internshipId,
      task_id: task.task_id,
      deliverable_type: deliverable,
      title: sentenceLabel(deliverable) + " - " + task.title,
      request_id: requestId("artifact"),
    });
    return result.artifact as InternshipWorkArtifact;
  }

  async function startText(deliverable: string) {
    setBusy("edit:" + deliverable); setActionError("");
    try {
      const artifact = await createArtifact(deliverable);
      let text = "";
      if (artifact?.current_version_id && artifact.current_source_type === "text") {
        const response = await fetch(
          API + "/artifacts/" + encodeURIComponent(artifact.id) + "/versions/" +
          encodeURIComponent(String(artifact.current_version_id)) + "/text?internship_id=" +
          encodeURIComponent(internshipId),
          { credentials: "include", cache: "no-store" },
        );
        if (response.ok) {
          const payload = await response.json() as { content?: string };
          text = payload.content || "";
        }
      }
      setDrafts((current) => ({ ...current, [deliverable]: text }));
      setEditing((current) => ({ ...current, [deliverable]: true }));
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "The draft could not be opened.");
    } finally { setBusy(""); }
  }

  async function saveText(deliverable: string) {
    setBusy("save:" + deliverable); setActionError("");
    try {
      const artifact = await createArtifact(deliverable);
      const latestReview = [...(artifact.reviews || [])].sort((a,b)=>b.created_at-a.created_at)[0];
      await callJson("/artifacts/" + encodeURIComponent(artifact.id) + "/versions/text", {
        internship_id: internshipId,
        content: drafts[deliverable] || "",
        request_id: requestId("version"),
        prior_review_id: latestReview?.decision === "changes_requested" ? latestReview.id : "",
      });
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "The draft could not be saved.");
    } finally { setBusy(""); }
  }

  async function upload(deliverable: string, file: File | null) {
    if (!file) return;
    setBusy("upload:" + deliverable); setActionError("");
    try {
      const artifact = await createArtifact(deliverable);
      const latestReview = [...(artifact.reviews || [])].sort((a,b)=>b.created_at-a.created_at)[0];
      const uploadRequestId = requestId("upload");
      const response = await fetch(
        API + "/artifacts/" + encodeURIComponent(artifact.id) + "/versions/upload?internship_id=" + encodeURIComponent(internshipId),
        {
          method: "POST",
          credentials: "include",
          headers: {
            "content-type": file.type || "application/octet-stream",
            "x-murikah-artifact-filename": encodeURIComponent(file.name),
            "x-murikah-request-id": uploadRequestId,
            "x-murikah-prior-review-id": latestReview?.decision === "changes_requested" ? latestReview.id : "",
          },
          body: file,
        },
      );
      if (!response.ok) throw new Error(await responseError(response, "The file could not be uploaded."));
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "The file could not be uploaded.");
    } finally { setBusy(""); }
  }

  async function submit(artifact: InternshipWorkArtifact) {
    if (!artifact.current_version_id) return;
    const confirmed = window.confirm(
      "Submit Version " + artifact.current_version_number +
      " for supervisor review? This version will remain in your submission history.",
    );
    if (!confirmed) return;
    setBusy("submit:" + artifact.id); setActionError("");
    try {
      await callJson("/artifacts/" + encodeURIComponent(artifact.id) + "/submit", {
        internship_id: internshipId,
        artifact_version_id: artifact.current_version_id,
        request_id: requestId("submit"),
      });
      setEditing((current)=>({...current,[artifact.deliverable_type]:false}));
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "The work could not be submitted.");
    } finally { setBusy(""); }
  }

  async function retryReview(artifact: InternshipWorkArtifact) {
    const latest = [...(artifact.submissions || [])].sort((a,b)=>b.submission_number-a.submission_number)[0];
    if (!latest || !["submitted","under_review"].includes(latest.status)) return;
    setBusy("review:" + artifact.id); setActionError("");
    try {
      await callJson("/submissions/" + encodeURIComponent(latest.id) + "/review", {
        internship_id: internshipId,
        request_id: requestId("review"),
      });
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "Supervisor review is temporarily unavailable.");
    } finally { setBusy(""); }
  }

  async function requestFormalAssessment(artifact: InternshipWorkArtifact) {
    const submissionId = artifact.accepted_submission_id || "";
    if (!submissionId) return;
    setBusy("assessment:" + artifact.id); setActionError("");
    try {
      const result = await callJson("/submissions/" + encodeURIComponent(submissionId) + "/assessment", {
        internship_id: internshipId,
        request_id: requestId("assessment"),
      });
      if (result.status === "failed" && result.message) setActionError(String(result.message));
      await onChanged();
    } catch (cause) {
      setActionError(cause instanceof Error ? cause.message : "Formal assessment is temporarily unavailable.");
    } finally { setBusy(""); }
  }

  return (
    <Panel className="p-4 sm:p-5" ariaLabel="Assignment detail">
      <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
        <div>
          <p className="text-xs font-semibold uppercase tracking-[0.1em] text-[#A9822E]">Assignment</p>
          <h2 className="mt-2 text-xl font-semibold">{task.title}</h2>
          <p className="mt-1 text-sm text-[var(--muted-foreground)]">
            {person?.name ? "Assigned by " + person.name : "Workplace assignment"}
          </p>
        </div>
        <button type="button" onClick={onClose} className="min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold">
          Close
        </button>
      </div>
      <dl className="mt-5 grid gap-5 sm:grid-cols-2">
        {[
          ["Assignment status", sentenceLabel(task.status)],
          ["Work status", sentenceLabel(task.work_status || "not_acknowledged")],
          ["Deadline", task.due_at ? formatDate(task.due_at, true) : "No deadline set"],
          ["Expected effort", task.expected_effort || "Not specified"],
          ["Difficulty", task.difficulty ? sentenceLabel(task.difficulty) : "Not specified"],
          ["Mentor support", task.mentor_support?.label || "Clarification only"],
        ].map(([term, description]) => (
          <div key={term}>
            <dt className="text-xs font-semibold uppercase tracking-[0.08em] text-[var(--muted-foreground)]">{term}</dt>
            <dd className="mt-1 text-sm leading-6">{description}</dd>
          </div>
        ))}
      </dl>
      {[
        ["Business context", task.business_context],
        ["Objective", task.learner_objective],
        ["Brief", task.brief],
      ].map(([title, copy]) => copy ? (
        <div key={title} className="mt-5">
          <h3 className="text-sm font-semibold">{title}</h3>
          <p className="mt-1 whitespace-pre-wrap text-sm leading-6 text-[var(--muted-foreground)]">{copy}</p>
        </div>
      ) : null)}

      {!task.acknowledged_at && task.status === "available" && !readOnly ? (
        <div className="mt-6 rounded-xl border border-[var(--border)] p-4">
          <h3 className="font-semibold">Ready to begin?</h3>
          <p className="mt-1 text-sm text-[var(--muted-foreground)]">Opening the assignment does not acknowledge it. Acknowledge it when you are ready to start work.</p>
          <button type="button" disabled={Boolean(busy)} onClick={() => void acknowledge()}
            className="mt-3 min-h-10 rounded-lg bg-[#1E2A30] px-4 text-sm font-semibold text-white disabled:opacity-60">
            {busy === "ack" ? "Acknowledging..." : "Acknowledge assignment"}
          </button>
        </div>
      ) : null}

      <div className="mt-6">
        <h3 className="text-base font-semibold">Work product / Deliverables</h3>
        <p className="mt-1 text-sm text-[var(--muted-foreground)]">
          Saving creates a new immutable version. Submitting is a separate action and keeps the submitted version in history.
        </p>
        {actionError ? <p className="mt-3 rounded-lg border border-red-300/60 px-3 py-2 text-sm text-red-700 dark:text-red-300" role="alert">{actionError}</p> : null}
        <div className="mt-4 space-y-4">
          {(task.deliverable_types || []).map((deliverable) => {
            const artifact = artifactFor(deliverable);
            const latestReview = artifact ? [...(artifact.reviews || [])].sort((a,b)=>b.created_at-a.created_at)[0] : undefined;
            const latestSubmission = artifact ? [...(artifact.submissions || [])].sort((a,b)=>b.submission_number-a.submission_number)[0] : undefined;
            const currentVersion = artifact?.current_version_id
              ? (artifact.versions || []).find((version) => version.id === artifact.current_version_id)
              : undefined;
            const canWork = Boolean(task.acknowledged_at) && task.status === "in_progress" && !readOnly &&
              (!artifact || ["draft","changes_requested"].includes(artifact.status));
            const canSubmit = Boolean(
              artifact?.current_version_id && canWork && !currentVersion?.submitted
            );
            const formalAssessments = [...(artifact?.formal_assessments || [])].sort((a,b)=>b.created_at-a.created_at);
            const acceptedAssessment = artifact?.accepted_submission_id
              ? formalAssessments.find((item)=>item.submission_id===artifact.accepted_submission_id)
              : undefined;
            const canAssess = Boolean(
              task.assessment_rubric && artifact?.accepted_submission_id &&
              !acceptedAssessment && !readOnly
            );
            return (
              <section key={deliverable} className="rounded-xl border border-[var(--border)] p-4">
                <div className="flex flex-col gap-2 sm:flex-row sm:items-start sm:justify-between">
                  <div>
                    <h4 className="font-semibold">{sentenceLabel(deliverable)}</h4>
                    <p className="mt-1 text-xs text-[var(--muted-foreground)]">
                      {artifact ? sentenceLabel(artifact.status) : "No draft yet"}
                      {artifact?.current_version_number ? " · Latest Version " + artifact.current_version_number : ""}
                    </p>
                    {artifact?.current_filename ? (
                      <p className="mt-1 text-xs text-[var(--muted-foreground)]">{artifact.current_filename}</p>
                    ) : null}
                  </div>
                  {artifact?.status === "accepted" ? (
                    <span className="w-fit rounded-full border border-emerald-300 px-2.5 py-1 text-xs font-semibold text-emerald-700 dark:text-emerald-300">Accepted</span>
                  ) : null}
                </div>

                {latestReview ? (
                  <div className="mt-4 rounded-lg border border-[var(--border)] bg-[var(--background)] p-3">
                    <p className="text-xs font-semibold uppercase tracking-[0.08em]">{sentenceLabel(latestReview.decision)}</p>
                    <p className="mt-2 whitespace-pre-wrap text-sm leading-6">{latestReview.feedback}</p>
                    {latestReview.requested_changes?.length ? (
                      <ul className="mt-2 list-disc space-y-1 pl-5 text-sm text-[var(--muted-foreground)]">
                        {latestReview.requested_changes.map((item) => <li key={item}>{item}</li>)}
                      </ul>
                    ) : null}
                    <p className="mt-2 text-xs text-[var(--muted-foreground)]">
                      {person?.name || "Simulated supervisor"} · {formatDate(latestReview.created_at, true)}
                    </p>
                  </div>
                ) : latestSubmission ? (
                  <div className="mt-4 rounded-lg border border-[var(--border)] p-3 text-sm">
                    Submitted for review · Submission {latestSubmission.submission_number} · {formatDate(latestSubmission.submitted_at, true)}
                  </div>
                ) : null}

                {textDeliverables.has(deliverable) && canWork ? (
                  <div className="mt-4">
                    {!editing[deliverable] ? (
                      <button type="button" disabled={Boolean(busy)} onClick={() => void startText(deliverable)}
                        className="min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold">
                        {artifact?.status === "changes_requested" ? "Create revision" : artifact?.current_version_number ? "Continue editing" : "Create draft"}
                      </button>
                    ) : (
                      <div>
                        <label htmlFor={"draft-" + deliverable} className="text-sm font-semibold">Draft</label>
                        <textarea id={"draft-" + deliverable} rows={10} value={drafts[deliverable] || ""}
                          onChange={(event) => setDrafts((current)=>({...current,[deliverable]:event.target.value}))}
                          className="mt-2 w-full rounded-lg border border-[var(--border)] bg-[var(--background)] p-3 text-sm leading-6 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#A9822E]"
                          placeholder="Draft your work here. Markdown is supported as plain text." />
                        <div className="mt-2 flex flex-wrap gap-2">
                          <button type="button" disabled={Boolean(busy) || !(drafts[deliverable] || "").trim()} onClick={() => void saveText(deliverable)}
                            className="min-h-10 rounded-lg bg-[#1E2A30] px-3 text-sm font-semibold text-white disabled:opacity-60">
                            {busy === "save:" + deliverable ? "Saving..." : "Save draft"}
                          </button>
                          <button type="button" onClick={() => setEditing((current)=>({...current,[deliverable]:false}))}
                            className="min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold">Close editor</button>
                        </div>
                      </div>
                    )}
                  </div>
                ) : null}

                {canWork ? (
                  <div className="mt-4">
                    <label htmlFor={"upload-" + deliverable} className="block text-sm font-semibold">Upload file</label>
                    <p className="mt-1 text-xs leading-5 text-[var(--muted-foreground)]">
                      Up to 10 MB. Supported: PDF, DOCX, XLSX, CSV, PPTX, TXT, Markdown, common images and safe text/code files. Uploaded code is stored as data and is never executed.
                    </p>
                    <input id={"upload-" + deliverable} type="file"
                      onChange={(event) => { const file = event.target.files?.[0] || null; void upload(deliverable,file); event.currentTarget.value = ""; }}
                      className="mt-2 block w-full text-sm file:mr-3 file:rounded-lg file:border file:border-[var(--border)] file:bg-transparent file:px-3 file:py-2 file:text-sm file:font-semibold"
                      aria-describedby={"upload-help-" + deliverable} />
                    <span id={"upload-help-" + deliverable} className="sr-only">Select one supported file up to 10 MB.</span>
                  </div>
                ) : null}

                {artifact && canSubmit ? (
                  <button type="button" disabled={Boolean(busy)} onClick={() => void submit(artifact)}
                    className="mt-4 min-h-10 rounded-lg bg-[#1E2A30] px-4 text-sm font-semibold text-white disabled:opacity-60">
                    {busy === "submit:" + artifact.id ? "Submitting..." : "Submit Version " + artifact.current_version_number + " for review"}
                  </button>
                ) : null}
                {latestSubmission && ["submitted","under_review"].includes(latestSubmission.status) && !readOnly ? (
                  <button type="button" disabled={Boolean(busy)} onClick={() => void retryReview(artifact!)}
                    className="mt-3 min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold disabled:opacity-60">
                    {busy === "review:" + artifact?.id ? "Requesting review..." : "Retry supervisor review"}
                  </button>
                ) : null}

                {canAssess ? (
                  <button type="button" disabled={Boolean(busy)} onClick={() => void requestFormalAssessment(artifact!)}
                    className="mt-3 min-h-10 rounded-lg border border-[#A9822E]/50 px-3 text-sm font-semibold disabled:opacity-60">
                    {busy === "assessment:" + artifact?.id ? "Assessing submitted work..." : "Request formal assessment"}
                  </button>
                ) : null}

                {acceptedAssessment ? (
                  <div className="mt-4 rounded-lg border border-[#A9822E]/30 bg-[#A9822E]/[0.04] p-4" aria-label="Formal assessment">
                    <div className="flex flex-wrap items-center justify-between gap-2">
                      <div>
                        <p className="text-xs font-semibold uppercase tracking-[0.08em] text-[#A9822E]">Formal assessment</p>
                        <p className="mt-1 text-sm font-semibold">
                          {task.assessment_rubric?.title || "Authored rubric assessment"}
                        </p>
                      </div>
                      <span className="rounded-full border border-[var(--border)] px-2.5 py-1 text-xs font-semibold">
                        {sentenceLabel(acceptedAssessment.status)}
                      </span>
                    </div>
                    {acceptedAssessment.completed_at ? (
                      <p className="mt-2 text-xs text-[var(--muted-foreground)]">
                        Assessed Version {(artifact.versions || []).find((v)=>v.id===acceptedAssessment.artifact_version_id)?.version_number || ""} · {formatDate(acceptedAssessment.completed_at,true)}
                      </p>
                    ) : null}
                    {acceptedAssessment.status === "completed" ? (
                      <>
                        {acceptedAssessment.overall_summary ? <p className="mt-3 text-sm leading-6">{acceptedAssessment.overall_summary}</p> : null}
                        <div className="mt-4 space-y-3">
                          {(acceptedAssessment.criteria || []).map((criterion) => {
                            const authored = task.assessment_rubric?.criteria.find((item)=>item.criterion_id===criterion.criterion_id);
                            return (
                              <article key={criterion.criterion_id} className="rounded-lg border border-[var(--border)] bg-[var(--background)] p-3">
                                <div className="flex flex-wrap items-center justify-between gap-2">
                                  <h5 className="text-sm font-semibold">{authored?.description || sentenceLabel(criterion.criterion_id)}</h5>
                                  <span className="text-xs font-semibold">{criterion.result_state === "not_assessed" ? "Not assessed" : sentenceLabel(criterion.rating_id)}</span>
                                </div>
                                {criterion.feedback ? <p className="mt-2 text-sm leading-6">{criterion.feedback}</p> : null}
                                {criterion.limitation ? <p className="mt-2 text-sm leading-6 text-[var(--muted-foreground)]">Limitation: {criterion.limitation}</p> : null}
                                {criterion.evidence_refs?.length ? (
                                  <div className="mt-3">
                                    <p className="text-xs font-semibold uppercase tracking-[0.06em] text-[var(--muted-foreground)]">Evidence</p>
                                    <ul className="mt-1 space-y-1 text-xs text-[var(--muted-foreground)]">
                                      {criterion.evidence_refs.map((ref) => (
                                        <li key={ref.evidence_ref}>
                                          {ref.locator?.kind === "line_range" ? "Lines " + ref.locator.start_line + "-" + ref.locator.end_line : sentenceLabel(ref.locator?.kind || "submitted evidence")}
                                        </li>
                                      ))}
                                    </ul>
                                  </div>
                                ) : null}
                              </article>
                            );
                          })}
                        </div>
                        {acceptedAssessment.limitations?.length ? (
                          <div className="mt-3 text-xs text-[var(--muted-foreground)]">
                            <span className="font-semibold">Assessment limitations: </span>{acceptedAssessment.limitations.join(" ")}
                          </div>
                        ) : null}
                      </>
                    ) : acceptedAssessment.status === "failed" ? (
                      <p className="mt-3 text-sm text-[var(--muted-foreground)]">No formal result was recorded. The submitted work and supervisor review remain unchanged.</p>
                    ) : (
                      <p className="mt-3 text-sm text-[var(--muted-foreground)]">Assessment is in progress. No result is shown until structured validation completes.</p>
                    )}
                  </div>
                ) : null}

                {artifact?.versions?.length ? (
                  <details className="mt-5">
                    <summary className="cursor-pointer text-sm font-semibold">Version and submission history</summary>
                    <div className="mt-3 space-y-2">
                      {[...artifact.versions].sort((a,b)=>b.version_number-a.version_number).map((version) => {
                        const submission = (artifact.submissions || []).find((item)=>item.artifact_version_id===version.id);
                        const review = submission ? (artifact.reviews || []).find((item)=>item.submission_id===submission.id) : undefined;
                        return (
                          <div key={version.id} className="rounded-lg border border-[var(--border)] p-3 text-sm">
                            <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
                              <div>
                                <p className="font-semibold">Version {version.version_number}</p>
                                <p className="mt-0.5 text-xs text-[var(--muted-foreground)]">
                                  {formatDate(version.created_at,true)}
                                  {version.original_filename && version.id !== artifact.current_version_id ? " · " + version.original_filename : ""}
                                  {version.size_bytes ? " · " + Math.ceil(version.size_bytes/1024) + " KB" : ""}
                                </p>
                              </div>
                              <a href={API + "/artifacts/" + encodeURIComponent(artifact.id) + "/versions/" + encodeURIComponent(version.id) + "/download?internship_id=" + encodeURIComponent(internshipId)}
                                className="inline-flex min-h-9 w-fit items-center rounded-lg border border-[var(--border)] px-3 text-xs font-semibold">
                                Download
                              </a>
                            </div>
                            {submission ? <p className="mt-2 text-xs">Submission {submission.submission_number}: {sentenceLabel(submission.status)}</p> : <p className="mt-2 text-xs text-[var(--muted-foreground)]">Not submitted</p>}
                            {review ? <p className="mt-1 text-xs text-[var(--muted-foreground)]">Review: {sentenceLabel(review.decision)}</p> : null}
                          </div>
                        );
                      })}
                    </div>
                  </details>
                ) : null}
              </section>
            );
          })}
        </div>
      </div>
    </Panel>
  );
}


function completionBlockerText(code: string) {
  const copy: Record<string, string> = {
    duration_not_met: "The minimum duration has not yet been reached.",
    required_tasks_incomplete: "Required work is still incomplete.",
    required_review_missing: "A required performance review is not finalized.",
    competency_evidence_insufficient: "One or more competency evidence requirements are not yet satisfied.",
    capstone_incomplete: "Required capstone work is still incomplete.",
    final_review_missing: "The final performance review is not finalized.",
    final_review_precedes_capstone: "The final performance review must be finalized after the required capstone work.",
    non_qualifying_internship: "This simulation is non-qualifying and cannot receive qualifying completion.",
    completion_policy_unavailable: "Completion requirements are not available for this pinned scenario version.",
    internship_stopped: "A stopped internship cannot be finalized as qualifying-complete.",
    internship_not_active: "This internship is no longer active.",
  };
  return copy[code] || sentenceLabel(code);
}

function CompletionRequirements({
  internshipId,
  onCompleted,
}: {
  internshipId: string;
  onCompleted: () => Promise<void>;
}) {
  const [status, setStatus] = useState<CompletionStatus | null>(null);
  const [loading, setLoading] = useState(true);
  const [finalizing, setFinalizing] = useState(false);
  const [message, setMessage] = useState("");

  async function loadCompletion() {
    setLoading(true);
    setMessage("");
    try {
      const response = await fetch(
        API + "/completion?internship_id=" + encodeURIComponent(internshipId),
        { credentials: "include", cache: "no-store" },
      );
      if (!response.ok) throw new Error(await responseError(response, "Completion requirements could not be loaded."));
      const payload = (await response.json()) as { completion?: CompletionStatus };
      setStatus(payload.completion || null);
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "Completion requirements could not be loaded.");
    } finally {
      setLoading(false);
    }
  }

  async function finalizeCompletion() {
    if (!status?.overall_eligible || status.lifecycle_status !== "active") return;
    setFinalizing(true);
    setMessage("");
    try {
      const response = await fetch(API + "/completion", {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          internship_id: internshipId,
          request_id: requestId("complete"),
        }),
      });
      if (!response.ok) throw new Error(await responseError(response, "The internship could not be completed."));
      const payload = (await response.json()) as { completion?: CompletionStatus };
      if (payload.completion) setStatus(payload.completion);
      await onCompleted();
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "The internship could not be completed.");
      await loadCompletion();
    } finally {
      setFinalizing(false);
    }
  }

  useEffect(() => { void loadCompletion(); }, [internshipId]);

  if (loading) {
    return (
      <Panel className="p-5" ariaLabel="Completion requirements">
        <p className="text-sm text-[var(--muted-foreground)]" role="status">Loading completion requirements…</p>
      </Panel>
    );
  }

  if (!status) {
    return (
      <Panel className="p-5" ariaLabel="Completion requirements">
        <h2 className="text-lg font-semibold">Completion requirements</h2>
        <p className="mt-2 text-sm text-red-600" role="alert">{message || "Completion requirements are unavailable."}</p>
      </Panel>
    );
  }

  const competencies = status.evidence.competencies || [];
  const completed = status.lifecycle_status === "completed";
  const durationDay = Math.min(status.duration.elapsed_days + 1, Math.max(1, status.duration.required_days));

  return (
    <Panel className="p-5 sm:p-6" ariaLabel="Completion requirements">
      <div className="flex flex-col justify-between gap-4 md:flex-row md:items-start">
        <div>
          <p className="text-xs font-semibold uppercase tracking-[0.12em] text-[#A9822E]">Internship status</p>
          <h2 className="mt-2 text-xl font-semibold">Completion requirements</h2>
          <p className="mt-2 max-w-3xl text-sm leading-6 text-[var(--muted-foreground)]">
            Qualifying completion is determined from server-authoritative duration, required work, finalized reviews,
            Competency Passport evidence and capstone requirements. Time alone does not complete the internship.
          </p>
        </div>
        {completed ? (
          <div className="shrink-0 rounded-lg border border-emerald-300 px-4 py-3 text-sm">
            <p className="font-semibold">Completed</p>
            <p className="mt-1 text-xs text-[var(--muted-foreground)]">
              {status.completed_at ? formatDate(status.completed_at, true) : "Completion recorded"}
            </p>
          </div>
        ) : null}
      </div>

      <div className="mt-5 grid gap-3 lg:grid-cols-2">
        <CompletionRequirementRow
          label="Duration requirement"
          detail={"Day " + durationDay + " of " + status.duration.required_days + ". Target date: " + formatDate(status.duration.target_end_at) + "."}
          passed={status.duration.passed}
        />
        <CompletionRequirementRow
          label="Required work"
          detail={status.required_work.completed + " of " + status.required_work.required + " required assignments completed."}
          passed={status.required_work.passed}
          href="/virtual-internship/work"
        />
        <CompletionRequirementRow
          label="Required performance reviews"
          detail={status.reviews.completed + " of " + status.reviews.required + " authored required reviews finalized."}
          passed={status.reviews.passed}
          href="/virtual-internship/activity"
        />
        <CompletionRequirementRow
          label="Competency evidence"
          detail={status.evidence.satisfied + " of " + status.evidence.required + " competency requirements satisfied."}
          passed={status.evidence.passed}
          href="/virtual-internship/passport"
        />
        <CompletionRequirementRow
          label="Capstone work"
          detail={status.capstone_final_review.capstones_completed + " of " + status.capstone_final_review.capstones_required + " required capstone assignments completed."}
          passed={status.capstone_final_review.capstones_completed === status.capstone_final_review.capstones_required}
          href="/virtual-internship/work"
        />
        <CompletionRequirementRow
          label="Final performance review"
          detail={
            status.capstone_final_review.final_review_required
              ? (status.capstone_final_review.final_review_finalized
                  ? "The required final performance review is finalized."
                  : "The required final performance review is pending.")
              : "No final performance review is required by this pinned completion policy."
          }
          passed={!status.capstone_final_review.final_review_required || status.capstone_final_review.final_review_finalized}
          href="/virtual-internship/activity"
        />
      </div>

      {competencies.length ? (
        <details open className="mt-4 rounded-lg border border-[var(--border)] p-4">
          <summary className="cursor-pointer text-sm font-semibold">Competency evidence requirements</summary>
          <div className="mt-3 space-y-3">
            {competencies.map((item) => (
              <div key={item.competency_id + ":" + item.definition_version} className="rounded-lg border border-[var(--border)] p-3">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <p className="text-sm font-semibold">{item.name}</p>
                  <span className="text-xs font-semibold">{item.passed ? "Met" : "Pending"}</span>
                </div>
                <p className="mt-1 text-xs leading-5 text-[var(--muted-foreground)]">
                  Level {sentenceLabel(item.current_level || "not demonstrated")} of required {sentenceLabel(item.minimum_level)}.
                  {" "}Evidence strength {sentenceLabel(item.current_evidence_strength || "none")} of required {sentenceLabel(item.minimum_evidence_strength)}.
                  {" "}{item.evidence_records} of {item.minimum_evidence_records} evidence records,
                  {" "}{item.independent_demonstrations} of {item.minimum_independent_demonstrations} independent demonstrations,
                  {" "}{item.distinct_contexts} of {item.minimum_distinct_contexts} distinct contexts.
                </p>
              </div>
            ))}
          </div>
        </details>
      ) : null}

      {status.blockers.length ? (
        <div className="mt-4 rounded-lg border border-amber-300/60 bg-amber-50/40 p-4 dark:bg-amber-950/10" role="status">
          <p className="text-sm font-semibold">What remains</p>
          <ul className="mt-2 list-disc space-y-1 pl-5 text-sm leading-6">
            {status.blockers.map((code) => <li key={code}>{completionBlockerText(code)}</li>)}
          </ul>
        </div>
      ) : null}

      {message ? <p className="mt-4 text-sm text-red-600" role="alert">{message}</p> : null}

      {completed ? (
        <p className="mt-4 text-xs leading-5 text-[var(--muted-foreground)]">
          This is a Murikah Virtual Internship simulation. Historical work, reviews and Competency Passport evidence remain available in read-only form.
        </p>
      ) : status.overall_eligible && status.lifecycle_status === "active" ? (
        <div className="mt-5">
          <button
            type="button"
            onClick={() => void finalizeCompletion()}
            disabled={finalizing}
            className="inline-flex min-h-10 items-center justify-center rounded-lg bg-[var(--foreground)] px-4 text-sm font-semibold text-[var(--background)] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#A9822E] disabled:opacity-60"
          >
            {finalizing ? "Completing…" : "Complete internship"}
          </button>
          <p className="mt-2 text-xs leading-5 text-[var(--muted-foreground)]">
            The server will re-check every requirement before recording completion.
          </p>
        </div>
      ) : null}
    </Panel>
  );
}

function renderDocumentContent(content: unknown) {
  if (typeof content === "string" || typeof content === "number" || typeof content === "boolean") {
    return <p className="whitespace-pre-wrap text-sm leading-7">{String(content)}</p>;
  }
  if (Array.isArray(content)) {
    return (
      <ul className="space-y-2 text-sm leading-6">
        {content.map((item, index) => <li key={index} className="rounded-lg border border-[var(--border)] p-3">{typeof item === "object" ? JSON.stringify(item) : String(item)}</li>)}
      </ul>
    );
  }
  if (content && typeof content === "object") {
    return (
      <dl className="space-y-3">
        {Object.entries(content as Record<string, unknown>).map(([key, value]) => (
          <div key={key} className="grid gap-1 sm:grid-cols-[180px_1fr]">
            <dt className="text-xs font-semibold text-[var(--muted-foreground)]">{sentenceLabel(key)}</dt>
            <dd className="text-sm leading-6">{typeof value === "object" ? JSON.stringify(value) : String(value ?? "")}</dd>
          </div>
        ))}
      </dl>
    );
  }
  return <p className="text-sm text-[var(--muted-foreground)]">This document has no displayable content.</p>;
}

function DocumentsView({
  documents,
  internshipId,
  completed,
}: {
  documents: InternshipDocument[];
  internshipId: string;
  completed: boolean;
}) {
  const [selected, setSelected] = useState<InternshipDocument | null>(null);
  const [loading, setLoading] = useState("");
  const [error, setError] = useState("");
  const [completionDocuments, setCompletionDocuments] = useState<CompletionDocument[]>([]);
  const [completionLoading, setCompletionLoading] = useState(completed);
  const [issuing, setIssuing] = useState("");
  const [completionError, setCompletionError] = useState("");

  const currentCompletionDocuments = completionDocuments.filter((document) => document.issuance_status === "current");

  async function loadCompletionDocuments() {
    if (!completed) {
      setCompletionDocuments([]);
      setCompletionLoading(false);
      return;
    }
    setCompletionLoading(true);
    try {
      const response = await fetch(
        API + "/completion-documents?internship_id=" + encodeURIComponent(internshipId),
        { credentials: "include", cache: "no-store" },
      );
      if (!response.ok) throw new Error(await responseError(response, "Completion documents could not be loaded."));
      const payload = (await response.json()) as { documents?: CompletionDocument[] };
      setCompletionDocuments(Array.isArray(payload.documents) ? payload.documents : []);
      setCompletionError("");
    } catch (cause) {
      setCompletionError(cause instanceof Error ? cause.message : "Completion documents could not be loaded.");
    } finally {
      setCompletionLoading(false);
    }
  }

  useEffect(() => {
    void loadCompletionDocuments();
  }, [completed, internshipId]);

  async function issueCompletionDocument(documentType: "performance_report" | "completion_letter") {
    setIssuing(documentType);
    setCompletionError("");
    try {
      const response = await fetch(API + "/completion-documents", {
        method: "POST",
        credentials: "include",
        cache: "no-store",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          internship_id: internshipId,
          document_type: documentType,
          request_id: requestId("completion-document"),
        }),
      });
      if (!response.ok) throw new Error(await responseError(response, "That completion document could not be issued."));
      await loadCompletionDocuments();
    } catch (cause) {
      setCompletionError(cause instanceof Error ? cause.message : "That completion document could not be issued.");
    } finally {
      setIssuing("");
    }
  }

  async function openDocument(document: InternshipDocument) {
    setLoading(document.document_id);
    setError("");
    try {
      const response = await fetch(API + "/documents/" + encodeURIComponent(document.document_id) + "?internship_id=" + encodeURIComponent(internshipId), {
        credentials: "include",
        cache: "no-store",
      });
      if (!response.ok) throw new Error(await responseError(response, "That document could not be loaded. Try again."));
      const payload = (await response.json()) as { document?: InternshipDocument };
      setSelected(payload.document || null);
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "That document could not be loaded. Try again.");
    } finally {
      setLoading("");
    }
  }

  function currentDocument(documentType: CompletionDocument["document_type"]) {
    return currentCompletionDocuments.find((document) => document.document_type === documentType);
  }

  return (
    <div className="space-y-5">
      <Panel className="p-5 sm:p-6" ariaLabel="Completion documents">
        <div className="flex flex-col justify-between gap-4 sm:flex-row sm:items-start">
          <div>
            <div className="flex items-center gap-2">
              <FileText size={19} />

function InboxView({
  workspace,
  onWorkspaceRefresh,
}: {
  workspace: InternshipWorkspaceSummary;
  onWorkspaceRefresh: () => Promise<void>;
}) {
  const router = useRouter();
  const searchParams = useSearchParams();
  const workplaceThreads = (workspace.threads || []).filter((thread) => thread.thread_kind === "workplace");
  const people = workspace.people || [];
  const requestedThread = searchParams.get("thread") || "";
  const requestedActor = searchParams.get("actor") || "";
  const initialThread = workplaceThreads.find((thread) => thread.thread_id === requestedThread) || null;
  const [selectedThread, setSelectedThread] = useState<InternshipThread | null>(initialThread);
  const [selectedActorId, setSelectedActorId] = useState(initialThread?.scenario_actor_id || requestedActor);
  const [messages, setMessages] = useState<InternshipMessage[]>([]);
  const [loading, setLoading] = useState(false);
  const [draft, setDraft] = useState("");
  const [sending, setSending] = useState(false);
  const [error, setError] = useState("");
  const [failed, setFailed] = useState<{ text: string; requestId: string; actorId: string; taskId: string } | null>(null);
  const internshipId = workspace.internship!.internship_id;
  const stopped = workspace.state === "stopped";
  const completed = workspace.state === "completed";
  const readOnly = stopped || completed;

  useEffect(() => {
    const thread = workplaceThreads.find((item) => item.thread_id === requestedThread) || null;
    if (thread) {
      setSelectedThread(thread);
      setSelectedActorId(thread.scenario_actor_id);
    } else if (requestedActor) {
      const actorThread = workplaceThreads.find((item) => item.scenario_actor_id === requestedActor) || null;
      setSelectedThread(actorThread);
      setSelectedActorId(requestedActor);
    }
  }, [requestedThread, requestedActor, workspace.threads]);

  useEffect(() => {
    if (!selectedThread) {
      setMessages([]);
      return;
    }
    let cancelled = false;
    setLoading(true);
    setError("");
    fetch(API + "/threads/" + encodeURIComponent(selectedThread.thread_id) + "?internship_id=" + encodeURIComponent(internshipId), {
      credentials: "include",
      cache: "no-store",
    })
      .then(async (response) => {
        if (!response.ok) throw new Error(await responseError(response, "This conversation could not be loaded. Try again."));
        return response.json();
      })
      .then((payload: { messages?: InternshipMessage[] }) => {
        if (!cancelled) setMessages(Array.isArray(payload.messages) ? payload.messages : []);
      })
      .catch((cause) => {
        if (!cancelled) setError(cause instanceof Error ? cause.message : "This conversation could not be loaded. Try again.");
      })
      .finally(() => {
        if (!cancelled) setLoading(false);
      });
    return () => { cancelled = true; };
  }, [selectedThread?.thread_id, internshipId]);

  function chooseThread(thread: InternshipThread) {
    setSelectedThread(thread);
    setSelectedActorId(thread.scenario_actor_id);
    router.push("/virtual-internship/inbox?thread=" + encodeURIComponent(thread.thread_id));
  }

  function choosePerson(person: InternshipPerson) {
    const existing = workplaceThreads.find((thread) => thread.scenario_actor_id === person.actor_id);
    if (existing) return chooseThread(existing);
    setSelectedThread(null);
    setSelectedActorId(person.actor_id);
    setMessages([]);
    router.push("/virtual-internship/inbox?actor=" + encodeURIComponent(person.actor_id));
  }

  async function sendWith(payload: { text: string; requestId: string; actorId: string; taskId: string }, optimistic: boolean) {
    if (!payload.actorId || !payload.text.trim() || readOnly) return;
    setSending(true);
    setError("");
    setFailed(null);
    if (optimistic) {
      setMessages((current) => current.concat({
        id: "optimistic-" + payload.requestId,
        sender_type: "learner",
        body: payload.text,
        created_at: Math.floor(Date.now() / 1000),
        request_id: payload.requestId,
      }));
      setDraft("");
    }
    let streamed = "";
    try {
      const response = await fetch(API + "/actor/messages/stream", {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          internship_id: internshipId,
          scenario_actor_id: payload.actorId,
          text: payload.text,
          request_id: payload.requestId,
          task_id: payload.taskId,
        }),
      });
      if (!response.ok) throw new Error(await responseError(response, "This workplace response is temporarily unavailable. Try again."));
      const replyId = "stream-" + payload.requestId;
      await readNdjson(response, (event) => {
        if (event.type === "chunk") {
          streamed += event.text || "";
          setMessages((current) => {
            const without = current.filter((message) => message.id !== replyId);
            return without.concat({ id: replyId, sender_type: "actor", body: streamed, created_at: Math.floor(Date.now() / 1000) });
          });
        }
        if (event.type === "error") throw new Error(event.message || "This workplace response is temporarily unavailable. Try again.");
      });
      await onWorkspaceRefresh();
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "This workplace response is temporarily unavailable. Try again.");
      setFailed(payload);
    } finally {
      setSending(false);
    }
  }

  function send() {
    void sendWith({ text: draft, requestId: requestId("actor"), actorId: selectedActorId, taskId: selectedThread?.related_task_id || "" }, true);
  }

  const person = people.find((item) => item.actor_id === selectedActorId);
  return (
    <div className="grid min-h-0 gap-5 md:grid-cols-[300px_minmax(0,1fr)]">
      <Panel className={(selectedActorId ? "hidden md:block " : "") + "p-3"}>
        <h2 className="px-2 py-2 text-base font-semibold">Workplace inbox</h2>
        <div className="mt-1 space-y-1">
          {workplaceThreads.map((thread) => (
            <button key={thread.thread_id} type="button" onClick={() => chooseThread(thread)} className={"w-full rounded-lg px-3 py-3 text-left text-sm " + (selectedThread?.thread_id === thread.thread_id ? "bg-[#A9822E]/10" : "hover:bg-[var(--background)]")}>
              <span className="block font-semibold">{thread.title}</span>
              <span className="mt-1 block truncate text-xs text-[var(--muted-foreground)]">{thread.last_message || "Conversation started"}</span>
            </button>
          ))}
          {!workplaceThreads.length ? <p className="px-3 py-4 text-sm text-[var(--muted-foreground)]">No workplace messages yet.</p> : null}
        </div>
        <h3 className="mt-4 px-2 text-xs font-semibold uppercase tracking-[0.08em] text-[var(--muted-foreground)]">People</h3>
        <div className="mt-1 space-y-1">
          {people.filter((item) => item.active).map((item) => (
            <button key={item.actor_id} type="button" onClick={() => choosePerson(item)} className="w-full rounded-lg px-3 py-2 text-left text-sm hover:bg-[var(--background)]">
              <span className="block font-medium">{item.name}</span>
              <span className="block text-xs text-[var(--muted-foreground)]">{item.job_title}</span>
            </button>
          ))}
        </div>
      </Panel>
      <div>
        {selectedActorId ? (
          <>
            <button type="button" onClick={() => { router.push("/virtual-internship/inbox"); setSelectedActorId(""); setSelectedThread(null); }} className="mb-3 min-h-10 rounded-lg border border-[var(--border)] px-3 text-sm font-semibold md:hidden">
              Back to inbox
            </button>
            <Conversation
              title={person?.name || selectedThread?.title || "Workplace conversation"}
              messages={messages}
              loading={loading}
              error={error}
              draft={draft}
              disabled={readOnly}
              sending={sending}
              onDraft={setDraft}
              onSend={send}
              onRetry={failed ? () => void sendWith(failed, false) : undefined}
            />
          </>
        ) : (
          <Panel className="p-5">
            <EmptyState title="Choose a workplace conversation." detail="Select a learner-visible person or an existing thread. Workplace messages are separate from ordinary Tutor chat." />
          </Panel>
        )}
      </div>
    </div>
  );
}

    return () => { cancelled = true; };
  }, [mentorThread?.thread_id, internshipId]);

  async function sendWith(payload: { text: string; requestId: string; taskId: string }, optimistic: boolean) {

    setSending(true);
    setError("");

          setMessages((current) => current.filter((message) => message.id !== replyId).concat({ id: replyId, sender_type: "mentor", body: streamed, created_at: Math.floor(Date.now() / 1000) }));
        }
        if (event.type === "error") throw new Error(event.message || "The Mentor is temporarily unavailable. Try again.");
      });
      await onWorkspaceRefresh();
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "The Mentor is temporarily unavailable. Try again.");
      setFailed(payload);
    } finally {
      setSending(false);
    }
  }

  function send() {
    void sendWith({ text: draft, requestId: requestId("mentor"), taskId }, true);
  }

  return (
    <div className="grid gap-5 lg:grid-cols-[minmax(0,1fr)_300px]">
      <Conversation
        title="Murikah Mentor"
        messages={messages}
        loading={loading}
        error={error}
        draft={draft}
        disabled={readOnly}
        sending={sending}
        onDraft={setDraft}
        onSend={send}
        onRetry={failed ? () => void sendWith(failed, false) : undefined}
        assistanceLabel={assistanceLabel || tasks.find((task) => task.task_id === taskId)?.mentor_support?.label}
        mentor
      />
      <Panel className="p-5">
        <h2 className="font-semibold">What the Mentor can help with</h2>
        <ul className="mt-3 space-y-2 text-sm leading-6 text-[var(--muted-foreground)]">
          <li>Explain a concept or workplace feedback.</li>
          <li>Help you plan how to approach an assignment.</li>
          <li>Coach a skill or ask reflective questions.</li>
          <li>Prepare with you within the permitted support level.</li>
        </ul>
        <label htmlFor="mentor-task-context" className="mt-5 block text-sm font-semibold">Assignment context</label>
        <select id="mentor-task-context" value={taskId} onChange={(event) => setTaskId(event.target.value)} className="mt-2 min-h-10 w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 text-sm">
          <option value="">General coaching</option>
          {tasks.map((task) => <option key={task.task_id} value={task.task_id}>{task.title}</option>)}
        </select>
        <p className="mt-4 text-xs leading-5 text-[var(--muted-foreground)]">The Mentor does not submit, complete or approve your work.</p>
      </Panel>
    </div>
  );
}

function ReflectionView({
  workspace,
  onWorkspaceRefresh,
}: {
  workspace: InternshipWorkspaceSummary;
  onWorkspaceRefresh: () => Promise<void>;
}) {
  const internshipId = workspace.internship!.internship_id;
  const periodKey = "week-" + workspace.internship!.current_week;
  const existing = (workspace.reflections || []).find((reflection) => reflection.period_key === periodKey);
  const [content, setContent] = useState(existing?.content || "");
  const [saving, setSaving] = useState(false);
  const [message, setMessage] = useState("");
  const readOnly = workspace.state !== "active";

  useEffect(() => setContent(existing?.content || ""), [existing?.reflection_id, existing?.content]);

  async function save() {
    if (!content.trim() || readOnly) return;
    setSaving(true);
    setMessage("");
    try {
      const response = await fetch(API + "/reflections", {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ internship_id: internshipId, period_key: periodKey, content: content.trim() }),
      });
      if (!response.ok) throw new Error(await responseError(response, "That reflection could not be saved. Try again."));
      setMessage("Reflection saved.");
      void onWorkspaceRefresh();
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "That reflection could not be saved. Try again.");
    } finally {
      setSaving(false);
    }
  }

  return (
    <Panel className="p-5">
      <h2 className="text-lg font-semibold">Reflection</h2>
      <p className="mt-1 text-sm text-[var(--muted-foreground)]">Week {workspace.internship!.current_week} learning journal</p>
      <div className="mt-4 rounded-lg bg-[var(--background)] p-4">
        <p className="text-sm font-semibold">Prompts you can use</p>
        <p className="mt-2 text-sm leading-6 text-[var(--muted-foreground)]">
          What did I work on? What was difficult? What did I learn? What feedback did I receive? What evidence supports my conclusion? What will I do differently next week?
        </p>
      </div>
      <label htmlFor="internship-reflection" className="mt-4 block text-sm font-semibold">Your reflection</label>
      <textarea id="internship-reflection" rows={8} value={content} onChange={(event) => setContent(event.target.value)} disabled={readOnly || saving} className="mt-2 w-full resize-y rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-sm leading-6 outline-none focus-visible:ring-2 focus-visible:ring-[#A9822E] disabled:opacity-60" />
      <div className="mt-3 flex flex-wrap items-center justify-between gap-3">
        <p className="text-xs text-[var(--muted-foreground)]" role="status">{message}</p>
        <button type="button" onClick={save} disabled={readOnly || saving || !content.trim()} className="min-h-10 rounded-lg bg-[#1E2A30] px-4 text-sm font-semibold text-white disabled:opacity-60">{saving ? "Saving…" : existing ? "Update reflection" : "Save reflection"}</button>
      </div>
    </Panel>
  );
}

function PerformanceReviews({
  workspace,
  refresh,
}: {
  workspace: InternshipWorkspaceSummary;
  refresh: () => Promise<void>;
}) {
  const [busy, setBusy] = useState("");
  const [error, setError] = useState("");
  const reviews = workspace.assessment?.performance_reviews || [];
  const eligibility = workspace.assessment?.review_eligibility;
  const readOnly = workspace.state !== "active";

  async function createReview(kind: "midpoint" | "final") {
    setBusy(kind); setError("");
    try {
      const response = await fetch(API + "/performance-reviews/" + kind, {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          internship_id: workspace.internship?.internship_id || "",
          request_id: requestId(kind + "-review"),
        }),
      });
      if (!response.ok) throw new Error(await responseError(response, "That performance review could not be created."));
      await refresh();
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "That performance review could not be created.");
    } finally { setBusy(""); }
  }

  return (
    <Panel className="p-5" ariaLabel="Performance reviews">
      <div className="flex flex-col gap-2 sm:flex-row sm:items-start sm:justify-between">
        <div>
          <h2 className="text-base font-semibold">Performance reviews</h2>
          <p className="mt-1 text-sm leading-6 text-[var(--muted-foreground)]">
            Midpoint and final performance reviews summarize durable internship evidence. They do not complete the internship or create a Competency Passport level.
          </p>
        </div>
        <div className="flex flex-wrap gap-2">
          {(["midpoint","final"] as const).map((kind) => {
            const existing = reviews.some((review)=>review.review_type===kind && review.status==="finalized");
            const allowed = Boolean(eligibility?.[kind]?.eligible) && !existing && !stopped;
            return (
              <button key={kind} type="button" disabled={!allowed || Boolean(busy)} onClick={() => void createReview(kind)}
                className="min-h-10 rounded-lg border border-[var(--border)] px-3 text-xs font-semibold disabled:opacity-50">
                {busy === kind ? "Creating..." : existing ? sentenceLabel(kind) + " review available" :
                  allowed ? "Create " + sentenceLabel(kind) + " review" :
                  sentenceLabel(kind) + " review from day " + String(eligibility?.[kind]?.required_day || "")}
              </button>
            );
          })}
        </div>
      </div>
      {error ? <p className="mt-3 text-sm text-red-600" role="alert">{error}</p> : null}
      <div className="mt-4 space-y-4">
        {reviews.length ? reviews.map((review) => (
          <article key={review.id} className="rounded-xl border border-[var(--border)] p-4">
            <div className="flex flex-wrap items-center justify-between gap-2">
              <h3 className="font-semibold">{review.review_type === "final" ? "Final Performance Review" : "Midpoint Performance Review"}</h3>
              <span className="text-xs text-[var(--muted-foreground)]">{formatDate(review.finalized_at || review.cutoff_at,true)}</span>
            </div>
            {review.narrative ? <p className="mt-3 text-sm leading-6">{review.narrative}</p> : null}
            <div className="mt-4 grid gap-4 md:grid-cols-3">
              {[
                ["Evidence-grounded strengths", review.strengths || []],
                ["Development areas", review.development_areas || []],
                ["Next-period priorities", review.priorities || []],
              ].map(([title, rows]) => (
                <div key={title as string}>
                  <h4 className="text-xs font-semibold uppercase tracking-[0.06em] text-[var(--muted-foreground)]">{title as string}</h4>
                  {(rows as Array<{text:string}>).length ? (
                    <ul className="mt-2 list-disc space-y-1 pl-4 text-sm leading-6">
                      {(rows as Array<{text:string}>).map((row,index)=><li key={index}>{row.text}</li>)}
                    </ul>
                  ) : <p className="mt-2 text-sm text-[var(--muted-foreground)]">No supported item recorded for this section.</p>}
                </div>
              ))}
            </div>
            <p className="mt-4 text-xs text-[var(--muted-foreground)]">
              Assistance context: {Object.entries(review.assistance_summary?.event_counts_by_level || {})
                .filter(([,count])=>Number(count)>0)
                .map(([level,count])=>"Level " + level + ": " + count)
                .join(" · ") || "No recorded assistance events before this review cutoff."}
            </p>
          </article>
        )) : (
          <p className="text-sm text-[var(--muted-foreground)]">No performance review has been finalized yet.</p>
        )}
      </div>
    </Panel>
  );
}

function passportRequirementLabel(item: { kind: string; count: number }) {
  const labels: Record<string, string> = {
    evidence_records: "additional qualifying evidence record",
    independent_demonstrations: "additional independent demonstration",
    distinct_task_contexts: "additional distinct task context",
    distinct_transfer_contexts: "additional distinct transfer context",
  };
  const label = labels[item.kind] || sentenceLabel(item.kind);
  return String(item.count) + " " + label + (item.count === 1 ? "" : "s");
}

function competencyRequirementSummary(level: string, rules: Record<string, number | string>) {
  const parts: string[] = [];
  const minRecords = Number(rules.min_records || 0);
  const minIndependent = Number(rules.min_independent || 0);
  const minTasks = Number(rules.min_task_contexts || 0);
  const minContexts = Number(rules.min_transfer_contexts || 0);
  if (minRecords) parts.push(String(minRecords) + " qualifying evidence record" + (minRecords === 1 ? "" : "s"));
  if (minIndependent) parts.push(String(minIndependent) + " independent demonstration" + (minIndependent === 1 ? "" : "s"));
  if (minTasks) parts.push(String(minTasks) + " distinct task context" + (minTasks === 1 ? "" : "s"));
  if (minContexts) parts.push(String(minContexts) + " distinct transfer context" + (minContexts === 1 ? "" : "s"));
  return sentenceLabel(level) + " requires " + (parts.length ? parts.join(", ") : "authored qualifying evidence") + ".";
}

function PassportView() {
  const [passport, setPassport] = useState<CompetencyPassportResponse | null>(null);
  const [loading, setLoading] = useState(true);
  const [message, setMessage] = useState("");
  const [selected, setSelected] = useState<CompetencyPassportItem | null>(null);
  const [evidence, setEvidence] = useState<CompetencyEvidenceView[]>([]);
  const [evidenceLoading, setEvidenceLoading] = useState(false);
  const [exporting, setExporting] = useState(false);
  const [includeNameInExport, setIncludeNameInExport] = useState(false);

  async function loadPassport() {
    setLoading(true);
    setMessage("");
    try {
      const response = await fetch(API + "/passport", { credentials: "include", cache: "no-store" });
      if (!response.ok) throw new Error(await responseError(response, "Your Competency Passport could not be loaded."));
      setPassport((await response.json()) as CompetencyPassportResponse);
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "Your Competency Passport could not be loaded.");
    } finally {
      setLoading(false);
    }
  }

  async function openEvidence(item: CompetencyPassportItem) {
    setSelected(item);
    setEvidence([]);
    setEvidenceLoading(true);
    setMessage("");
    try {
      const response = await fetch(API + "/passport/evidence?competency_id=" + encodeURIComponent(item.competency_id), {
        credentials: "include",
        cache: "no-store",
      });
      if (!response.ok) throw new Error(await responseError(response, "Competency evidence could not be loaded."));
      const payload = (await response.json()) as { evidence?: CompetencyEvidenceView[] };
      setEvidence(Array.isArray(payload.evidence) ? payload.evidence : []);
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "Competency evidence could not be loaded.");
    } finally {
      setEvidenceLoading(false);
    }
  }

  async function exportPassport() {
    setExporting(true);
    setMessage("");
    try {
      const response = await fetch(API + "/passport/export", {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ include_display_name: includeNameInExport }),
      });
      if (!response.ok) throw new Error(await responseError(response, "Your Passport export could not be created."));
      const blob = await response.blob();
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = "murikah-competency-passport.json";
      document.body.appendChild(link);
      link.click();
      link.remove();
      URL.revokeObjectURL(url);
    } catch (cause) {
      setMessage(cause instanceof Error ? cause.message : "Your Passport export could not be created.");
    } finally {
      setExporting(false);
    }
  }

  useEffect(() => { void loadPassport(); }, []);

  if (loading) {
    return <Panel className="p-6"><p className="text-sm text-[var(--muted-foreground)]" role="status">Loading your evidence-backed Competency Passport…</p></Panel>;
  }
  if (!passport) {
    return <Panel className="p-6"><p className="text-sm text-red-600" role="alert">{message || "Your Competency Passport is unavailable."}</p></Panel>;
  }

  const competencies = passport.competencies || [];
  const levelLabel = (id: string) => passport.level_framework?.find((level) => level.id === id)?.label || sentenceLabel(id);

  return (
    <div className="space-y-5">
      <Panel className="p-5 sm:p-6">
        <div className="flex flex-col justify-between gap-4 md:flex-row md:items-start">
          <div>
            <p className="text-xs font-semibold uppercase tracking-[0.12em] text-[#A9822E]">Verified simulation evidence</p>
            <h2 className="mt-2 text-xl font-semibold">Competency Passport</h2>
            <p className="mt-2 max-w-3xl text-sm leading-6 text-[var(--muted-foreground)]">
              This private record summarizes competencies demonstrated through formally assessed Murikah Virtual Internship work.
              Levels are calculated from traceable evidence, not an AI opinion or an overall employability score.
            </p>
          </div>
          <div className="flex shrink-0 flex-col items-start gap-2 md:items-end">
            <label className="flex items-center gap-2 text-xs text-[var(--muted-foreground)]">
              <input
                type="checkbox"
                checked={includeNameInExport}
                onChange={(event) => setIncludeNameInExport(event.target.checked)}
              />
              Include my display name in this export
            </label>
            <button
              type="button"
              onClick={() => void exportPassport()}
              disabled={exporting}
              className="inline-flex min-h-10 items-center justify-center rounded-lg border border-[var(--border)] px-4 text-sm font-semibold hover:bg-[var(--background)] disabled:opacity-60"
            >
              {exporting ? "Preparing export…" : "Export JSON"}
            </button>
          </div>
        </div>
        {message ? <p className="mt-4 text-sm text-red-600" role="alert">{message}</p> : null}
      </Panel>

      {competencies.length === 0 ? (
        <Panel className="p-5">
          <EmptyState
            title="No verified competency evidence yet."
            detail="Your Passport will grow as assessed internship work creates verified evidence. Starting or submitting work alone does not award a competency level."
          />
          <details className="mt-5 rounded-lg border border-[var(--border)] p-4">
            <summary className="cursor-pointer text-sm font-semibold">What the Passport measures</summary>
            <div className="mt-4 grid gap-3 md:grid-cols-2">
              {(passport.definitions || []).map((definition) => (
                <div key={definition.competency_id} className="rounded-lg border border-[var(--border)] p-3">
                  <p className="text-sm font-semibold">{definition.name}</p>
                  <p className="mt-1 text-xs leading-5 text-[var(--muted-foreground)]">{definition.description}</p>
                  {definition.context_metadata?.physical ? (
                    <p className="mt-2 text-xs leading-5 text-[var(--muted-foreground)]">Virtual Internship evidence cannot fully verify the physical/manual component of this competency.</p>
                  ) : null}
                  <ul className="mt-2 space-y-1 text-xs leading-5 text-[var(--muted-foreground)]">
                    {Object.entries(definition.evidence_requirements || {}).map(([level, rules]) => (
                      <li key={level}>• {competencyRequirementSummary(level, rules)}</li>
                    ))}
                  </ul>
                </div>
              ))}
            </div>
          </details>
        </Panel>
      ) : (
        <div className="grid gap-4 xl:grid-cols-2">
          {competencies.map((item) => (
            <Panel key={item.competency_id + ":" + item.definition_version} className="p-5">
              <div className="flex flex-wrap items-start justify-between gap-3">
                <div>
                  <p className="text-xs font-semibold uppercase tracking-[0.08em] text-[var(--muted-foreground)]">{sentenceLabel(item.domain)}</p>
                  <h3 className="mt-1 text-lg font-semibold">{item.name}</h3>
                </div>
                <span className="rounded-full border border-[var(--border)] px-3 py-1 text-xs font-semibold">
                  {levelLabel(item.current_level)}
                </span>
              </div>
              <p className="mt-3 text-sm leading-6 text-[var(--muted-foreground)]">{item.description}</p>
              <dl className="mt-4 grid grid-cols-2 gap-3 text-sm sm:grid-cols-4">
                <div><dt className="text-xs text-[var(--muted-foreground)]">Evidence</dt><dd className="mt-1 font-semibold">{item.evidence_count}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Strength</dt><dd className="mt-1 font-semibold">{sentenceLabel(item.evidence_strength_summary)}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Independent</dt><dd className="mt-1 font-semibold">{item.independent_count}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Assisted</dt><dd className="mt-1 font-semibold">{item.assisted_count}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Task contexts</dt><dd className="mt-1 font-semibold">{item.distinct_task_count}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Transfer contexts</dt><dd className="mt-1 font-semibold">{item.distinct_context_count}</dd></div>
                <div><dt className="text-xs text-[var(--muted-foreground)]">Internships</dt><dd className="mt-1 font-semibold">{item.distinct_internship_count}</dd></div>
                {item.trend !== "insufficient_evidence" ? (
                  <div><dt className="text-xs text-[var(--muted-foreground)]">Trend</dt><dd className="mt-1 font-semibold">{sentenceLabel(item.trend)}</dd></div>
                ) : null}
              </dl>
              <p className="mt-4 text-xs leading-5 text-[var(--muted-foreground)]">
                Why this level: {item.explanation?.qualifying_evidence_records ?? item.evidence_count} qualifying evidence record{(item.explanation?.qualifying_evidence_records ?? item.evidence_count) === 1 ? "" : "s"},
                {" "}{item.explanation?.independent_demonstrations ?? item.independent_count} independent demonstration{(item.explanation?.independent_demonstrations ?? item.independent_count) === 1 ? "" : "s"},
                across {item.explanation?.distinct_task_contexts ?? item.distinct_task_count} task context{(item.explanation?.distinct_task_contexts ?? item.distinct_task_count) === 1 ? "" : "s"} and

                    <div><dt className="font-semibold">Evidence strength</dt><dd className="mt-0.5 text-[var(--muted-foreground)]">{sentenceLabel(row.evidence_strength)}</dd></div>
                    <div><dt className="font-semibold">Assistance context</dt><dd className="mt-0.5 text-[var(--muted-foreground)]">{row.assistance_context?.label || "Recorded assistance context"}</dd></div>
                    <div><dt className="font-semibold">Assessment date</dt><dd className="mt-0.5 text-[var(--muted-foreground)]">{formatDate(row.assessment_date || row.created_at, true)}</dd></div>
                    <div><dt className="font-semibold">Internship</dt><dd className="mt-0.5 break-all text-[var(--muted-foreground)]">{row.internship_id}</dd></div>
                    <div><dt className="font-semibold">Task</dt><dd className="mt-0.5 break-all text-[var(--muted-foreground)]">{row.task_id}</dd></div>
                    <div><dt className="font-semibold">Work context</dt><dd className="mt-0.5 text-[var(--muted-foreground)]">{sentenceLabel(row.transfer_context?.work_context || "not specified")}</dd></div>
                    <div><dt className="font-semibold">Criterion</dt><dd className="mt-0.5 break-all text-[var(--muted-foreground)]">{row.criterion_id}</dd></div>
                    <div><dt className="font-semibold">Mapping version</dt><dd className="mt-0.5 text-[var(--muted-foreground)]">v{row.mapping_version}</dd></div>
                    <div><dt className="font-semibold">Artifact version</dt><dd className="mt-0.5 break-all text-[var(--muted-foreground)]">{row.artifact_version_id}</dd></div>
                    <div><dt className="font-semibold">Submission</dt><dd className="mt-0.5 break-all text-[var(--muted-foreground)]">{row.submission_id}</dd></div>
                  </dl>
                  <p className="mt-3 text-xs text-[var(--muted-foreground)]">
                    Evidence ID <span className="break-all font-mono">{row.id}</span> · Assessment <span className="break-all font-mono">{row.assessment_id}</span>
                  </p>
                  {Object.entries(row.strength_factors || {}).some(([, value]) => value === true) ? (
                    <div className="mt-3">
                      <p className="text-xs font-semibold">Why this evidence has this strength</p>
                      <ul className="mt-1 space-y-1 text-xs text-[var(--muted-foreground)]">
                        {Object.entries(row.strength_factors || {})
                          .filter(([, value]) => value === true)
                          .map(([factor]) => <li key={factor}>• {sentenceLabel(factor)}</li>)}
                      </ul>
                    </div>
                  ) : null}
                  {row.limitations?.length ? (
                    <div className="mt-3">
                      <p className="text-xs font-semibold">Limitations</p>
                      <ul className="mt-1 space-y-1 text-xs text-[var(--muted-foreground)]">{row.limitations.map((limitation, index) => <li key={index}>• {limitation}</li>)}</ul>
                    </div>
                  ) : null}
                  {row.adjustments?.length ? <p className="mt-3 text-xs text-amber-700 dark:text-amber-300">This evidence has an administrative audit adjustment and is excluded from the current aggregate where applicable.</p> : null}
                  <a href={"/virtual-internship/work?task=" + encodeURIComponent(row.task_id)} className="mt-4 inline-flex min-h-9 items-center rounded-lg border border-[var(--border)] px-3 text-xs font-semibold">
                    Open related work
                  </a>
                </article>
              )) : <EmptyState title="No evidence is available for this competency." detail="Only valid completed Phase 6 assessments with an authored competency mapping appear here." />}
            </div>
          )}
        </Panel>
      ) : null}
    </div>
  );
}

function ActivityView({
  workspace,
  refresh,
}: {
  workspace: InternshipWorkspaceSummary;
  refresh: () => Promise<void>;
}) {
  return (
    <div className="grid gap-5 lg:grid-cols-[minmax(0,0.9fr)_minmax(0,1.1fr)]">
      <Timeline title="Activity" items={workspace.activity || []} />
      <div className="space-y-5">
        <PerformanceReviews workspace={workspace} refresh={refresh} />
        <ReflectionView workspace={workspace} onWorkspaceRefresh={refresh} />
        <Panel className="p-5">
          <h2 className="text-base font-semibold">Past reflections</h2>
          <div className="mt-3 space-y-3">
            {(workspace.reflections || []).length ? (workspace.reflections || []).map((reflection) => (
              <article key={reflection.reflection_id} className="rounded-lg border border-[var(--border)] p-3">
                <p className="text-xs font-semibold uppercase tracking-[0.08em] text-[var(--muted-foreground)]">{sentenceLabel(reflection.period_key)}</p>
                <p className="mt-2 line-clamp-4 whitespace-pre-wrap text-sm leading-6">{reflection.content}</p>
              </article>
            )) : <p className="text-sm text-[var(--muted-foreground)]">No reflections saved yet.</p>}
          </div>
        </Panel>
      </div>
    </div>
  );
}

export default function MurikahVirtualInternshipWorkspace() {
  const pathname = usePathname();
  const router = useRouter();
  const [workspace, setWorkspace] = useState<InternshipWorkspaceSummary | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [starting, setStarting] = useState("");
  const [selectedTask, setSelectedTask] = useState<InternshipTaskSummary | null>(null);

  const section = useMemo(() => {
    const suffix = pathname.replace(/^\/virtual-internship\/?/, "").split("/")[0];
    return SECTIONS.some((item) => item.id === suffix) ? suffix : "overview";
  }, [pathname]);

  async function loadWorkspace() {
    setLoading(true);
    setError("");
    try {
      const response = await fetch(API + "/workspace", { credentials: "include", cache: "no-store" });
      if (!response.ok) throw new Error(await responseError(response, "We could not load your internship right now. Try again."));
      setWorkspace((await response.json()) as InternshipWorkspaceSummary);
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "We could not load your internship right now. Try again.");
    } finally {
      setLoading(false);
    }
  }

  async function refreshWorkspace() {
    try {
      const response = await fetch(API + "/workspace", { credentials: "include", cache: "no-store" });
      if (!response.ok) throw new Error(await responseError(response, "We could not refresh your internship right now."));
      setWorkspace((await response.json()) as InternshipWorkspaceSummary);
    } catch {
      // Preserve the durable view already on screen when a targeted refresh fails.
    }
  }

  useEffect(() => { void loadWorkspace(); }, []);

  async function startInternship(scenarioSlug: string) {
    setStarting(scenarioSlug);
    setError("");
    try {
      const response = await fetch(API + "/start", {
        method: "POST",
        credentials: "include",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ scenario_slug: scenarioSlug, request_id: requestId("start") }),
      });
      if (!response.ok) throw new Error(await responseError(response, "We could not start that internship right now. Try again."));
      setWorkspace((await response.json()) as InternshipWorkspaceSummary);
      router.push("/virtual-internship");
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "We could not start that internship right now. Try again.");
    } finally {
      setStarting("");
    }
  }

  if (loading) return <LoadingWorkspace />;
  if (error && !workspace) return <ErrorWorkspace message={error} retry={() => void loadWorkspace()} />;
  if (!workspace) return <ErrorWorkspace message="We could not load your internship right now. Try again." retry={() => void loadWorkspace()} />;
  if (workspace.state === "guest") return <GuestPreview />;
  if (workspace.state === "none") return <NoInternship options={workspace.start_options || []} onStart={(slug) => void startInternship(slug)} starting={starting} error={error} />;

  const internship = workspace.internship!;
  const company = workspace.company!;
  const people = workspace.people || [];
  const tasks = workspace.tasks || [];
  const stopped = workspace.state === "stopped";
  const completed = workspace.state === "completed";
  const readOnly = stopped || completed;
  const selectedTaskCurrent = selectedTask
    ? tasks.find((task) => task.task_id === selectedTask.task_id) || selectedTask
    : null;

  let body: React.ReactNode;
  if (section === "inbox") body = <InboxView workspace={workspace} onWorkspaceRefresh={refreshWorkspace} />;
  else if (section === "work") body = <WorkView tasks={tasks} people={people} internshipId={internship.internship_id} readOnly={readOnly} selected={selectedTaskCurrent} onSelect={setSelectedTask} onChanged={refreshWorkspace} />;
  else if (section === "company") body = <CompanyView company={company} people={people} onMessage={(person) => router.push("/virtual-internship/inbox?actor=" + encodeURIComponent(person.actor_id))} />;
  else if (section === "documents") body = <DocumentsView documents={workspace.documents || []} internshipId={internship.internship_id} completed={completed} />;
  else if (section === "meetings") body = <MeetingsView meetings={workspace.meetings || []} activity={workspace.activity || []} />;
  else if (section === "mentor") body = <MentorView workspace={workspace} onWorkspaceRefresh={refreshWorkspace} />;
  else if (section === "passport") body = <PassportView />;
  else if (section === "activity") body = <ActivityView workspace={workspace} refresh={refreshWorkspace} />;
  else body = <OverviewView workspace={workspace} people={people} onTask={setSelectedTask} onWorkspaceRefresh={refreshWorkspace} />;

  return (
    <main className="min-h-full overflow-x-hidden bg-[var(--background)] px-3 py-5 text-[var(--foreground)] sm:px-5 lg:px-8">
      <div className="mx-auto w-full max-w-7xl">
        <header>
          <div className="flex flex-col justify-between gap-4 lg:flex-row lg:items-start">
            <div className="min-w-0">
              <div className="flex flex-wrap items-center gap-2 text-xs font-semibold uppercase tracking-[0.12em] text-[#A9822E]">
                <span>Murikah Virtual Internship</span>
                <span aria-hidden="true">·</span>
                <span>Simulated workplace</span>
              </div>
              <h1 className="mt-2 text-2xl font-semibold tracking-[-0.025em] sm:text-3xl">{company.name || internship.scenario_title || "Virtual Internship"}</h1>
              <p className="mt-1 text-sm text-[var(--muted-foreground)]">{internship.role_title}</p>
            </div>
            <div className="flex flex-wrap items-center gap-2">
              <span className={"rounded-full border px-3 py-1.5 text-xs font-semibold " + (stopped ? "border-red-300 text-red-700 dark:text-red-300" : "border-emerald-300 text-emerald-700 dark:text-emerald-300")}>
                {completed ? "Completed" : stopped ? "Stopped" : "Active"}
              </span>
              <span className="rounded-full border border-[var(--border)] px-3 py-1.5 text-xs text-[var(--muted-foreground)]">
                Scenario v{internship.scenario_version}
              </span>
            </div>
          </div>
          {stopped ? (
            <div className="mt-4 rounded-lg border border-amber-300/60 bg-amber-50/50 px-4 py-3 text-sm leading-6 dark:bg-amber-950/10" role="status">
              This internship is stopped. Your history remains available, including artifact versions, submissions and reviews, but new workplace messages, Mentor requests, reflections and work changes are read-only.
            </div>
          ) : completed ? (
            <div className="mt-4 rounded-lg border border-emerald-300/60 bg-emerald-50/50 px-4 py-3 text-sm leading-6 dark:bg-emerald-950/10" role="status">
              This Murikah Virtual Internship simulation is completed. Historical work, messages, reviews, reflections and Competency Passport evidence remain available in read-only form.
            </div>
          ) : null}
          {error ? <p className="mt-3 text-sm text-red-600" role="alert">{error}</p> : null}
          <nav aria-label="Virtual Internship workspace" className="mt-6 overflow-x-auto overscroll-x-contain pb-1">
            <div className="flex min-w-max gap-1 rounded-xl border border-[var(--border)] bg-[var(--card)] p-1">
              {SECTIONS.map(({ id, label, icon: Icon, href }) => (
                <a key={id} href={href} aria-current={section === id ? "page" : undefined} className={"inline-flex min-h-10 items-center gap-2 rounded-lg px-3 text-sm font-medium focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#A9822E] " + (section === id ? "bg-[#A9822E]/10 text-[var(--foreground)]" : "text-[var(--muted-foreground)] hover:bg-[var(--background)]")}>
                  <Icon size={15} />
                  {label}
                </a>
              ))}
            </div>
          </nav>
        </header>
        <div className="mt-5">
          {section === "overview" && selectedTaskCurrent ? <div className="mb-5"><TaskDetail task={selectedTaskCurrent} people={people} internshipId={internship.internship_id} readOnly={readOnly} onChanged={refreshWorkspace} onClose={() => setSelectedTask(null)} /></div> : null}
          {body}
        </div>
      </div>
    </main>
  );
}
