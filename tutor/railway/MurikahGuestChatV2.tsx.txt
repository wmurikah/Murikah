function createHandoffId(): string {
  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
    return crypto.randomUUID();
  }
  return `guest-${Date.now()}-${Math.random().toString(36).slice(2)}`;
}

function buildConfiguration(spaceId: SpaceId, overrides: ConfigState): ConfigState {
  const configuration: ConfigState = {};
  for (const section of SPACE_CONFIGURATION[spaceId]) {
    for (const field of section.fields) {
      configuration[field.key] = overrides[field.key] ?? field.defaultValue;
    }
  }
  return configuration;
}

function isModeId(value: unknown): value is ModeId {
  return MODES.some((mode) => mode.id === value);
}

function isSpaceId(value: unknown): value is SpaceId {
  return SPACES.some((space) => space.id === value);
}

function SidebarSpaceButton({
  option,
  active,
  collapsed,
  onSelect,
}: {
  option: SpaceOption;
  active: boolean;
  collapsed: boolean;
  onSelect: () => void;
}) {
  const Icon = option.icon;
  return (
    <button
      type="button"
      onClick={onSelect}
      title={collapsed ? `${option.label} — ${option.description}` : option.description}
      aria-pressed={active}
      className={`group relative flex w-full items-center rounded-xl px-3 py-2.5 text-left text-[14px] font-medium transition-colors ${
        collapsed ? "justify-center" : "gap-3"
      } ${
        active
          ? "bg-[var(--muted)] text-[var(--foreground)]"
          : "text-[var(--foreground)]/78 hover:bg-[var(--muted)]/70 hover:text-[var(--foreground)]"
      }`}
    >
      <Icon size={18} strokeWidth={1.8} className="shrink-0" />
      {!collapsed && <span className="truncate">{option.label}</span>}
    </button>
  );
}

function SpaceConfigurator({
  space,
  values,
  onChange,
}: {
  space: SpaceOption;
  values: ConfigState;
  onChange: (key: string, value: ConfigValue) => void;
}) {
  const sections = SPACE_CONFIGURATION[space.id];
  if (!sections.length) return null;

  return (
    <div className="mb-4 overflow-hidden rounded-2xl border border-[var(--border)] bg-[var(--card)] shadow-sm">
      <div className="border-b border-[var(--border)]/70 px-4 py-3 sm:px-5">
        <div className="text-sm font-semibold text-[var(--foreground)]">Configure {space.label}</div>
        <p className="mt-1 max-w-2xl text-xs leading-5 text-[var(--muted-foreground)]">
          These guest controls are live. Changing them does not consume one of your seven learning interactions.
        </p>
      </div>
      <div className="space-y-4 p-4 sm:p-5">
        {sections.map((section) => (
          <section key={section.title} className="rounded-xl border border-[var(--border)]/80 p-3.5 sm:p-4">
            <div className="mb-3">
              <h3 className="text-[13px] font-semibold text-[var(--foreground)]">{section.title}</h3>
              {section.description && (
                <p className="mt-0.5 text-[11.5px] leading-5 text-[var(--muted-foreground)]">{section.description}</p>
              )}
            </div>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
              {section.fields.map((field) => {
                const value = values[field.key] ?? field.defaultValue;
                if (field.kind === "toggle") {
                  const enabled = Boolean(value);
                  return (
                    <button
                      key={field.key}
                      type="button"
                      aria-pressed={enabled}
                      onClick={() => onChange(field.key, !enabled)}
                      className="flex min-h-14 items-center justify-between gap-3 rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-left hover:bg-[var(--muted)]/50"
                    >
                      <span className="text-xs font-medium text-[var(--foreground)]">{field.label}</span>
                      <span aria-hidden className={`relative h-5 w-9 shrink-0 rounded-full transition-colors ${enabled ? "bg-[#A9822E]" : "bg-[var(--muted)]"}`}>
                        <span className={`absolute top-0.5 h-4 w-4 rounded-full bg-white shadow-sm transition-transform ${enabled ? "translate-x-[18px]" : "translate-x-0.5"}`} />
                      </span>
                    </button>
                  );
                }
                if (field.kind === "select") {
                  return (
                    <label key={field.key} className="block">
                      <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                      <select
                        value={String(value)}
                        onChange={(event) => onChange(field.key, event.target.value)}
                        className="w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs text-[var(--foreground)] outline-none focus:border-[#A9822E]"
                      >
                        {field.options.map((option) => <option key={option} value={option}>{option}</option>)}
                      </select>
                    </label>
                  );
                }
                if (field.kind === "textarea") {
                  return (
                    <label key={field.key} className="block sm:col-span-2">
                      <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                      <textarea
                        rows={3}
                        value={String(value)}
                        placeholder={field.placeholder}
                        onChange={(event) => onChange(field.key, event.target.value)}
                        className="w-full resize-y rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs leading-5 text-[var(--foreground)] outline-none focus:border-[#A9822E]"
                      />
                    </label>
                  );
                }
                return (
                  <label key={field.key} className="block">
                    <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                    <input
                      value={String(value)}
                      placeholder={field.placeholder}
                      onChange={(event) => onChange(field.key, event.target.value)}
                      className="w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs text-[var(--foreground)] outline-none focus:border-[#A9822E]"
                    />
                  </label>
                );
              })}
            </div>
          </section>
        ))}
      </div>
    </div>
  );
}

        if (typeof stored.handoffId === "string" && stored.handoffId) {
          handoffIdRef.current = stored.handoffId;
        }
      }
    } catch {
      window.localStorage.removeItem(GUEST_STATE_KEY);
    } finally {
      setHydrated(true);
    }
  }, []);

  const persistGuestState = useCallback(() => {
    if (!hydrated) return;
    const state: StoredGuestState = {
      version: 2,
      handoffId: handoffIdRef.current,
      messages: messages.slice(-24),
      modeId,
      spaceId,
      spaceConfigOverrides,
      persona,
      llmSelection,
      updatedAt: Date.now(),
    };
    try {
      window.localStorage.setItem(GUEST_STATE_KEY, JSON.stringify(state));
    } catch {
      // The chat remains usable even if local storage is unavailable.
    }
  }, [hydrated, llmSelection, messages, modeId, persona, spaceConfigOverrides, spaceId]);

  useEffect(() => {
    persistGuestState();
  }, [persistGuestState]);

  useEffect(() => {
    if (!hydrated) return;
    let alive = true;
    Promise.all([
      fetch("/api/auth/status", { cache: "no-store" }).then(async (response) =>
        response.ok ? ((await response.json()) as AuthStatusResponse) : null,
      ),
      fetch("/api/murikah/guest-status", { cache: "no-store" }).then(async (response) =>
        response.ok ? ((await response.json()) as GuestStatusResponse) : null,
      ),
      fetch("/api/murikah/guest-models", { cache: "no-store" }).then(async (response) =>
        response.ok ? ((await response.json()) as GuestModelsResponse) : null,
      ),
    ])
      .then(([auth, guest, models]) => {
        if (!alive) return;
        if (Array.isArray(models?.options)) setModelOptions(models.options);
        if (auth?.authenticated) {
          router.replace(messages.length ? RESUME_NEXT : "/chat");
          return;
        }
        if (guest) {
          setRemaining(Math.max(0, Number(guest.remaining ?? 7)));
          setAuthNeeded(Boolean(guest.requires_auth));
        }
        setChecking(false);
      })
      .catch(() => {
        if (alive) setChecking(false);
      });
    return () => {
      alive = false;
    };
  }, [hydrated, messages.length, router]);

  useEffect(() => {
    endRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
  }, [messages, busy, authNeeded]);

  useEffect(
    () => () => {
      recognitionRef.current?.stop();
    },
    [],
  );

  function updateSpaceConfiguration(key: string, value: ConfigValue) {
    setSpaceConfigOverrides((current) => ({
      ...current,
      [spaceId]: {
        ...current[spaceId],
        [key]: value,

  async function handleFiles(event: ChangeEvent<HTMLInputElement>) {
    const files = Array.from(event.target.files || []).slice(0, 4);
    event.target.value = "";
    if (!files.length) return;
    setError("");
    const accepted: GuestAttachment[] = [];
    for (const file of files) {
      if (file.size > 500_000) {
        setError(`${file.name} is too large for guest upload. Keep text files below 500 KB.`);
        continue;
      }
      const textLike =
        file.type.startsWith("text/") ||
        /\.(txt|md|csv|json|js|jsx|ts|tsx|py|html|css|xml|yaml|yml|log)$/i.test(file.name);
      if (!textLike) {
        setError("Guest upload currently reads text-based files. Sign in for PDF, Office and image workflows.");
        continue;
      }
      const text = (await file.text()).trim().slice(0, 12_000);
      if (text) accepted.push({ name: file.name, text });
    }
    if (accepted.length) {
      setAttachments((current) => [...current, ...accepted].slice(0, 4));
    }
  }

  function toggleVoice() {
    if (listening) {
      recognitionRef.current?.stop();
      setListening(false);
      return;
    }
    const speechWindow = window as typeof window & {
      SpeechRecognition?: SpeechRecognitionConstructor;
      webkitSpeechRecognition?: SpeechRecognitionConstructor;
    };
    const Recognition = speechWindow.SpeechRecognition || speechWindow.webkitSpeechRecognition;
    if (!Recognition) {
      setError("Voice input is not supported by this browser.");
      return;
    }
    const recognition = new Recognition();
    recognition.lang = "en-US";
    recognition.interimResults = false;
    recognition.continuous = false;
    recognition.onresult = (event) => {
      const transcript = Array.from(event.results)
        .map((result) => result[0]?.transcript || "")
        .join(" ")
        .trim();
      if (transcript) {
        setDraft((current) => `${current}${current.trim() ? " " : ""}${transcript}`);
      }
    };
    recognition.onend = () => setListening(false);
    recognition.onerror = () => {
      setListening(false);
      setError("Voice input stopped before a transcript was received.");
    };
    recognitionRef.current = recognition;
    setListening(true);
    recognition.start();
  }

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const prompt = draft.trim();
    if (!prompt || busy) return;
    if (remaining <= 0 || authNeeded) {
      setAuthNeeded(true);
      return;
    }

    const prior = messages.slice(-12);
    setMessages((current) => [...current, { role: "user", content: prompt }]);
    setDraft("");
    setBusy(true);
    setError("");

    try {
      const response = await fetch("/api/murikah/guest-chat", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          prompt,
          history: prior,
          mode: activeMode.id,
          space: activeSpace.id,
          configuration: activeConfiguration,
          persona,
          llm_selection: llmSelection,
          attachments,
        }),
      });
      const data = (await response.json().catch(() => ({}))) as GuestChatResponse;
      if (!response.ok) {
        if (response.status === 403) {
          setRemaining(0);
          setAuthNeeded(true);
          return;
        }
        throw new Error(
          typeof data.detail === "string" ? data.detail : "Tutor could not answer that prompt.",
        );
      }
      setMessages((current) => [
        ...current,
        { role: "assistant", content: String(data.answer || "") },
      ]);
      setAttachments([]);
      const nextRemaining = Math.max(0, Number(data.remaining) || 0);
      setRemaining(nextRemaining);
      setAuthNeeded(Boolean(data.requires_auth) || nextRemaining === 0);
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "Tutor could not answer that prompt.");
    } finally {
      setBusy(false);
    }
  }

  if (checking || !hydrated) {
    return (
      <main className="grid min-h-screen place-items-center bg-[var(--background)] px-6 text-sm text-[var(--muted-foreground)]">
        <div className="flex items-center gap-3">
          <span className="h-4 w-4 animate-spin rounded-full border-2 border-[#A9822E]/30 border-t-[#A9822E]" />
          Opening Murikah Tutor…
        </div>
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-[var(--background)] text-[var(--foreground)]">
      <div className="flex min-h-screen">
        <aside
          className={`sticky top-0 hidden h-screen shrink-0 flex-col border-r border-[var(--border)]/70 bg-[var(--background)] py-4 transition-[width] duration-200 lg:flex ${sidebarCollapsed ? "w-20 px-2" : "w-72 px-3"}`}
        >
          <div className={`mb-3 flex items-center ${sidebarCollapsed ? "justify-center" : "justify-between gap-2 px-2"}`}>
            <div
              title="Murikah Tutor"
              className={`inline-flex items-center rounded-lg bg-[#1E2A30] py-1.5 text-sm font-semibold tracking-tight text-white shadow-sm ${sidebarCollapsed ? "h-9 w-9 justify-center px-0" : "gap-2.5 px-3"}`}
            >
              <span>{sidebarCollapsed ? "M" : "Murikah"}</span>
              {!sidebarCollapsed && <span aria-hidden className="h-4 w-px bg-[#A9822E]" />}
              {!sidebarCollapsed && <span>Tutor</span>}
            </div>
            {!sidebarCollapsed && (
              <button
                type="button"
                onClick={() => setSidebarCollapsed(true)}
                title="Collapse sidebar"
                className="rounded-lg p-2 text-[var(--muted-foreground)] hover:bg-[var(--muted)] hover:text-[var(--foreground)]"
              >
                <PanelLeftClose size={17} />
              </button>
            )}
          </div>

          {sidebarCollapsed && (
            <button
              type="button"
              onClick={() => setSidebarCollapsed(false)}
              title="Expand sidebar"
              className="mx-auto mb-3 rounded-lg p-2 text-[var(--muted-foreground)] hover:bg-[var(--muted)] hover:text-[var(--foreground)]"
            >
              <PanelLeftOpen size={18} />
            </button>
          )}

          <div className="min-h-0 flex-1 overflow-y-auto pr-0.5">
            {!sidebarCollapsed && (
              <div className="px-3 pb-2 pt-1 text-[10px] font-semibold uppercase tracking-[0.14em] text-[var(--muted-foreground)]">
                Learning spaces
              </div>
            )}
            <nav className="space-y-1" aria-label="Learning spaces">
              {SPACES.map((option) => (
                <SidebarSpaceButton
                  key={option.id}
                  option={option}
                  active={option.id === spaceId}
                  collapsed={sidebarCollapsed}
                  onSelect={() => setSpaceId(option.id)}
                />
              ))}
            </nav>

            <div className="my-3 border-t border-[var(--border)]/70" />

            {sidebarCollapsed ? (
              <button
                type="button"
                onClick={() => {
                  setSidebarCollapsed(false);
                  setLearningModeOpen(true);
                }}
                title={`Choose Learning Mode — ${activeMode.label}`}
                className="flex w-full justify-center rounded-xl bg-[#A9822E]/10 p-3 text-[#A9822E] hover:bg-[#A9822E]/15"
              >
                <ActiveModeIcon size={19} />
              </button>
            ) : (
              <div>
                <button
                  type="button"
                  onClick={() => setLearningModeOpen((open) => !open)}
                  aria-expanded={learningModeOpen}
                  className="flex w-full items-center gap-2 rounded-xl px-3 py-2.5 text-left text-[13px] font-semibold text-[var(--foreground)] hover:bg-[var(--muted)]/70"
                >
                  <GraduationCap size={17} className="shrink-0 text-[#A9822E]" />
                  <span className="min-w-0 flex-1">Choose Learning Mode</span>
                  <ChevronRight size={15} className={`transition-transform ${learningModeOpen ? "rotate-90" : ""}`} />
                </button>
                {learningModeOpen && (
                  <div className="mt-1 space-y-1 pb-2" aria-label="Choose Learning Mode">
                    {MODES.map((mode) => {
                      const Icon = mode.icon;
                      const active = mode.id === modeId;
                      return (

                          type="button"
                          onClick={() => setModeId(mode.id)}
                          className={`flex w-full items-start gap-2.5 rounded-xl px-3 py-2 text-left transition-colors ${active ? "bg-[var(--muted)]" : "hover:bg-[var(--muted)]/60"}`}

                              {mode.label}
                              {active && <Sparkles size={11} className="text-[#A9822E]" />}
                            </span>

          </div>

          <div className="mt-3 border-t border-[var(--border)]/70 px-2 pt-3">
            <Link
              href={USERNAME_SIGN_IN}
              onClick={persistGuestState}
              title="Sign in and continue this conversation"
              className={`block rounded-lg py-2 text-sm font-medium text-[var(--foreground)]/75 hover:bg-[var(--muted)] hover:text-[var(--foreground)] ${sidebarCollapsed ? "px-1 text-center text-xs" : "px-3"}`}
            >
              {sidebarCollapsed ? "Sign in" : "Sign in · keep this chat"}
            </Link>
            {!sidebarCollapsed && <p className="mt-2 px-3 text-[11px] text-[var(--muted-foreground)]">Apache-2.0</p>}

            <div className="flex h-16 items-center justify-between px-4 sm:px-6">
              <div className="inline-flex items-center gap-2.5 rounded-lg bg-[#1E2A30] px-3 py-1.5 text-sm font-semibold tracking-tight text-white shadow-sm lg:hidden">
                <span>Murikah</span><span aria-hidden className="h-4 w-px bg-[#A9822E]" /><span>Tutor</span>
              </div>
              <div className="hidden items-center gap-2 text-sm font-medium text-[var(--muted-foreground)] lg:flex">

                const active = option.id === spaceId;

                <p className="mx-auto mt-4 max-w-2xl text-[15px] leading-7 text-[var(--muted-foreground)] sm:text-base">
                  Explore the Tutor before signing in. Your seven guest interactions can use learning modes, uploads, persona, voice and model selection — interface changes do not reduce the allowance.
                </p>
              </div>
            ) : messages.length === 0 ? (
              <div className="mx-auto w-full max-w-3xl pb-4 text-left">
                <p className="text-xs font-semibold uppercase tracking-[0.16em] text-[#A9822E]">Try {activeSpace.label}</p>
                <h1 className="mt-2 text-2xl font-semibold tracking-[-0.025em] sm:text-3xl">Set it up the way you want to learn.</h1>

                {messages.map((message, index) => (
                  <article
                    key={`${message.role}-${index}`}
                    className={message.role === "user" ? "ml-auto max-w-[85%] rounded-2xl bg-[var(--muted)] px-4 py-3" : "max-w-full"}
                  >
                    {message.role === "assistant" && (
                      <div className="mb-2 text-xs font-semibold text-[#A9822E]">Murikah Tutor · {activeMode.label}</div>
                    )}
                    <div className="whitespace-pre-wrap text-[15px] leading-7">{message.content}</div>
                  </article>

                {busy && <div className="text-sm text-[var(--muted-foreground)]">Murikah Tutor is thinking…</div>}
                <div ref={endRef} />
              </div>
            )}

            {authNeeded ? (

                <div className="text-center">
                  <h2 className="text-xl font-semibold tracking-tight">Sign in to keep learning</h2>

                    You have reached the seven-interaction guest limit. Your guest conversation is retained in this browser and will be continued in your account after sign-in — you will not lose the chat or start again.

                  <MurikahSocialButtons next={RESUME_NEXT} />
                </div>
                <div className="my-4 flex items-center gap-3 text-xs text-[var(--muted-foreground)] before:h-px before:flex-1 before:bg-[var(--border)] after:h-px after:flex-1 after:bg-[var(--border)]">or</div>

                  <span className="mt-0.5 rounded-lg bg-[#A9822E]/10 p-2 text-[#A9822E]"><ActiveSpaceIcon size={16} /></span>
                  <div className="min-w-0 flex-1">
                    <div className="text-xs font-semibold text-[var(--foreground)]">{activeSpace.label}</div>
                    <p className="mt-0.5 text-xs leading-5 text-[var(--muted-foreground)]">{activeSpace.description}</p>
                  </div>
                </div>

                {preferencesOpen && (
                  <div className="mb-3 grid gap-3 rounded-xl border border-[var(--border)] bg-[var(--card)] p-4 shadow-sm sm:grid-cols-2">
                    <label className="sm:col-span-2">
                      <span className="mb-1 block text-xs font-semibold">Tutor persona / personalisation</span>

                        className="w-full resize-y rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs leading-5 outline-none focus:border-[#A9822E]"

                  <textarea
                    value={draft}
                    onChange={(event) => setDraft(event.target.value)}
                    onKeyDown={(event) => {

                                  className={`flex w-full items-start gap-3 rounded-xl px-3 py-2.5 text-left transition-colors ${active ? "bg-[var(--muted)]" : "hover:bg-[var(--muted)]/70"}`}
                                >
                                  <Icon size={18} className="mt-0.5 shrink-0 text-[#A9822E]" />
                                  <span className="min-w-0 flex-1">
                                    <span className="flex items-center gap-2 text-sm font-semibold text-[var(--foreground)]">{mode.label}{active && <Sparkles size={13} className="text-[#A9822E]" />}</span>
                                    <span className="mt-0.5 block text-xs leading-5 text-[var(--muted-foreground)]">{mode.description}</span>
                                  </span>
                                </button>
                              );
                            })}
                          </div>
                        )}
                      </div>

                      <button type="button" onClick={() => fileInputRef.current?.click()} title="Upload text file" className="rounded-lg p-2 text-[var(--muted-foreground)] hover:bg-[var(--muted)] hover:text-[var(--foreground)]">
                        <Upload size={17} />
                      </button>
                      <button type="button" onClick={toggleVoice} title={listening ? "Stop voice input" : "Voice input"} className={`rounded-lg p-2 hover:bg-[var(--muted)] ${listening ? "text-[#A9822E]" : "text-[var(--muted-foreground)] hover:text-[var(--foreground)]"}`}>
                        {listening ? <MicOff size={17} /> : <Mic size={17} />}
                      </button>
                      <button type="button" onClick={() => setPreferencesOpen((open) => !open)} title="Persona and model" className={`inline-flex items-center gap-1.5 rounded-lg px-2 py-1.5 text-xs ${preferencesOpen ? "bg-[var(--muted)] text-[var(--foreground)]" : "text-[var(--muted-foreground)] hover:bg-[var(--muted)] hover:text-[var(--foreground)]"}`}>
                        <Settings2 size={16} />
                        <span className="hidden sm:inline">Personalise</span>
                      </button>
                      <span title={`Model: ${selectedModelLabel}`} className="hidden max-w-44 items-center gap-1.5 truncate rounded-lg px-2 py-1.5 text-[11px] text-[var(--muted-foreground)] md:inline-flex">
                        <Cpu size={14} />{selectedModelLabel}
                      </span>
                    </div>

                    <button type="submit" disabled={busy || !draft.trim()} className="rounded-lg bg-[#1E2A30] px-4 py-2 text-sm font-semibold text-white disabled:cursor-not-allowed disabled:opacity-40">
                      Send
                    </button>
                  </div>
                </div>
                <p className="mt-3 text-center text-[11px] text-[var(--muted-foreground)]">
                  {remaining} of 7 guest interactions remaining · UI controls do not count · your chat can continue after sign-in
                </p>
              </form>
            )}
          </section>
        </div>
      </div>
    </main>
  );
}
