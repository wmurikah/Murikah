"use client";

import { useMemo, useState } from "react";
import { Check, Copy, Mail, MessageCircle, Share2, X } from "lucide-react";

export default function MurikahInviteFriends({
  compact = false,
  className = "",
}: {
  compact?: boolean;
  className?: string;
}) {
  const [open, setOpen] = useState(false);
  const [copied, setCopied] = useState(false);
  const inviteUrl = useMemo(
    () =>
      typeof window === "undefined"
        ? "https://tutor.murikah.com/"
        : window.location.origin + "/",
    [],
  );
  const message = "Try Murikah Tutor with me: " + inviteUrl;
  const emailHref = "mailto:?subject=" + encodeURIComponent("Try Murikah Tutor") + "&body=" + encodeURIComponent(message);
  const whatsappHref = "https://wa.me/?text=" + encodeURIComponent(message);

  async function copyInvite() {
    try {
      await navigator.clipboard.writeText(inviteUrl);
      setCopied(true);
      window.setTimeout(() => setCopied(false), 1800);
    } catch {
      window.prompt("Copy this Murikah Tutor link:", inviteUrl);
    }
  }

  async function nativeShare() {
    if (navigator.share) {
      try {
        await navigator.share({
          title: "Murikah Tutor",
          text: "Try Murikah Tutor with me.",
          url: inviteUrl,
        });
        return;
      } catch {
        // Keep the modal open if the browser share sheet is dismissed.
      }
    }
    await copyInvite();
  }

  return (
    <>
      <button
        type="button"
        onClick={() => setOpen(true)}
        className={(compact ? "inline-flex size-9 items-center justify-center rounded-lg" : "inline-flex items-center gap-2 rounded-lg px-3 py-2 text-sm font-medium") + " border border-[var(--border)] bg-[var(--background)] text-[var(--foreground)]/80 transition-colors hover:bg-[var(--muted)] hover:text-[var(--foreground)] " + className}
        aria-label="Invite friends"
        title="Invite friends"
      >
        <Share2 size={16} />
        {!compact && <span>Invite friends</span>}
      </button>

      {open && (
        <div
          className="fixed inset-0 z-[90] grid place-items-center bg-black/35 px-4 backdrop-blur-[2px]"
          role="dialog"
          aria-modal="true"
          aria-labelledby="invite-friends-title"
          onMouseDown={(event) => {
            if (event.target === event.currentTarget) setOpen(false);
          }}
        >
          <div className="w-full max-w-md rounded-2xl border border-[var(--border)] bg-[var(--background)] p-5 shadow-2xl">
            <div className="flex items-start justify-between gap-4">
              <div>
                <h2 id="invite-friends-title" className="text-lg font-semibold">
                  Invite friends to Murikah Tutor
                </h2>
                <p className="mt-1 text-sm leading-6 text-[var(--muted-foreground)]">
                  Share Tutor directly. Your conversations and account data are never included in the invite.
                </p>
              </div>
              <button
                type="button"
                onClick={() => setOpen(false)}
                className="rounded-lg p-2 text-[var(--muted-foreground)] hover:bg-[var(--muted)]"
                aria-label="Close invite dialog"
              >
                <X size={17} />
              </button>
            </div>

            <div className="mt-5 flex items-center gap-2 rounded-xl border border-[var(--border)] bg-[var(--muted)]/35 p-2">
              <input
                readOnly
                value={inviteUrl}
                aria-label="Invite link"
                className="min-w-0 flex-1 bg-transparent px-2 text-sm outline-none"
              />
              <button
                type="button"
                onClick={copyInvite}
                className="inline-flex items-center gap-1.5 rounded-lg bg-[#1E2A30] px-3 py-2 text-xs font-semibold text-white"
              >
                {copied ? <Check size={14} /> : <Copy size={14} />}
                {copied ? "Copied" : "Copy link"}
              </button>
            </div>

            <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
              <a
                href={emailHref}
                className="inline-flex items-center justify-center gap-2 rounded-xl border border-[var(--border)] px-3 py-3 text-sm font-medium hover:bg-[var(--muted)]"
              >
                <Mail size={16} /> Email
              </a>
              <a
                href={whatsappHref}
                target="_blank"
                rel="noreferrer"
                className="inline-flex items-center justify-center gap-2 rounded-xl border border-[var(--border)] px-3 py-3 text-sm font-medium hover:bg-[var(--muted)]"
              >
                <MessageCircle size={16} /> WhatsApp
              </a>
              <button
                type="button"
                onClick={nativeShare}
                className="col-span-2 inline-flex items-center justify-center gap-2 rounded-xl border border-[var(--border)] px-3 py-3 text-sm font-medium hover:bg-[var(--muted)] sm:col-span-1"
              >
                <Share2 size={16} /> Share
              </button>
            </div>
          </div>
        </div>
      )}
    </>
  );
}
