"use client";
import { useEffect, useState } from "react";

let sessionRequest: Promise<Response> | null = null;

export default function MurikahWorkspaceEntry() {
  const [error, setError] = useState("");
  useEffect(() => {
    let alive = true;
    async function enter() {
      try {
        sessionRequest ??= fetch("/api/murikah/access/session", { method: "POST", credentials: "same-origin" });
        const res = await sessionRequest;
        if (!res.ok) throw new Error("Tutor couldn't open your workspace. Please try again.");
        const candidate = new URLSearchParams(window.location.search).get("next") || "/chat";
        const next = candidate.startsWith("/") && !candidate.startsWith("//") && !candidate.includes("\\") && candidate !== "/" ? candidate : "/chat";
        if (alive) window.location.replace(next);
      } catch (e) { if (alive) setError(e instanceof Error ? e.message : "Please try again."); }
    }
    void enter();
    return () => { alive = false; };
  }, []);
  return <main className="flex min-h-dvh items-center justify-center p-8"><div className="text-center">
    <h1 className="text-2xl font-semibold">Murikah Tutor</h1>
    <p className="mt-4" role="status">{error || "Opening your learning workspace…"}</p>
    {error && <button className="mt-4 rounded-lg bg-[#1E2A30] px-5 py-2 text-white" onClick={() => window.location.reload()}>Try again</button>}
    <p className="mt-5"><a href="/register">Sign up</a><span className="mx-3">·</span><a href="/login">Sign in</a></p>
  </div></main>;
}
