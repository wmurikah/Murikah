function SidebarButton({
  option,
  active,
  onSelect,
}: {
  option: SpaceOption;
  active: boolean;
  onSelect: () => void;
}) {
  const Icon = option.icon;
  return (
    <button
      type="button"
      onClick={onSelect}
      title={option.description}
      aria-pressed={active}
      className={`group relative flex w-full items-center gap-3 rounded-xl px-3 py-2.5 text-left text-[14px] font-medium transition-colors ${
        active
          ? "bg-[var(--muted)] text-[var(--foreground)]"
          : "text-[var(--foreground)]/78 hover:bg-[var(--muted)]/70 hover:text-[var(--foreground)]"
      }`}
    >
      <Icon size={18} strokeWidth={1.8} className="shrink-0" />
      <span className="truncate">{option.label}</span>
      <span className="pointer-events-none absolute left-[calc(100%+10px)] top-1/2 z-40 hidden w-64 -translate-y-1/2 rounded-xl border border-[var(--border)] bg-[var(--card)] px-3 py-2 text-xs font-normal leading-5 text-[var(--muted-foreground)] shadow-lg group-hover:block group-focus-visible:block">
        {option.description}
      </span>
    </button>
  );
}

function buildConfiguration(spaceId: SpaceId, overrides: ConfigState): ConfigState {
  const configuration: ConfigState = {};
  for (const section of SPACE_CONFIGURATION[spaceId]) {
    for (const field of section.fields) {
      configuration[field.key] = overrides[field.key] ?? field.defaultValue;
    }
  }
  return configuration;
}

function SpaceConfigurator({
  space,
  values,
  onChange,
}: {
  space: SpaceOption;
  values: ConfigState;
  onChange: (key: string, value: ConfigValue) => void;
}) {
  const sections = SPACE_CONFIGURATION[space.id];
  if (sections.length === 0) return null;

  return (
    <div className="mb-4 overflow-hidden rounded-2xl border border-[var(--border)] bg-[var(--card)] shadow-sm">
      <div className="border-b border-[var(--border)]/70 px-4 py-3 sm:px-5">
        <div className="flex items-start justify-between gap-4">
          <div>
            <div className="text-sm font-semibold text-[var(--foreground)]">Configure {space.label}</div>
            <p className="mt-1 max-w-2xl text-xs leading-5 text-[var(--muted-foreground)]">
              Try the configuration before creating an account. Changes apply to your next guest interaction; sign in only when you want to save persistent settings, accounts or private assets.
            </p>
          </div>
          <span className="hidden rounded-full border border-[#A9822E]/30 bg-[#A9822E]/10 px-2.5 py-1 text-[10px] font-semibold uppercase tracking-[0.12em] text-[#A9822E] sm:inline-flex">
            Interactive preview
          </span>
        </div>
      </div>

      <div className="space-y-4 p-4 sm:p-5">
        {sections.map((section) => (
          <section key={section.title} className="rounded-xl border border-[var(--border)]/80 p-3.5 sm:p-4">
            <div className="mb-3">
              <h3 className="text-[13px] font-semibold text-[var(--foreground)]">{section.title}</h3>
              {section.description && (
                <p className="mt-0.5 text-[11.5px] leading-5 text-[var(--muted-foreground)]">{section.description}</p>
              )}
            </div>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
              {section.fields.map((field) => {
                const value = values[field.key] ?? field.defaultValue;
                if (field.kind === "toggle") {
                  const enabled = Boolean(value);
                  return (
                    <button
                      key={field.key}
                      type="button"
                      aria-pressed={enabled}
                      onClick={() => onChange(field.key, !enabled)}
                      className="flex min-h-14 items-center justify-between gap-3 rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-left hover:bg-[var(--muted)]/50"
                    >
                      <span className="min-w-0">
                        <span className="block text-xs font-medium text-[var(--foreground)]">{field.label}</span>
                        {field.help && <span className="mt-0.5 block text-[10.5px] text-[var(--muted-foreground)]">{field.help}</span>}
                      </span>
                      <span
                        aria-hidden
                        className={`relative h-5 w-9 shrink-0 rounded-full transition-colors ${enabled ? "bg-[#A9822E]" : "bg-[var(--muted)]"}`}
                      >
                        <span
                          className={`absolute top-0.5 h-4 w-4 rounded-full bg-white shadow-sm transition-transform ${enabled ? "translate-x-[18px]" : "translate-x-0.5"}`}
                        />
                      </span>
                    </button>
                  );
                }

                if (field.kind === "select") {
                  return (
                    <label key={field.key} className="block">
                      <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                      <select
                        value={String(value)}
                        onChange={(event) => onChange(field.key, event.target.value)}
                        className="w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs text-[var(--foreground)] outline-none focus:border-[#A9822E]"
                      >
                        {field.options.map((option) => (
                          <option key={option} value={option}>{option}</option>
                        ))}
                      </select>
                      {field.help && <span className="mt-1 block text-[10.5px] text-[var(--muted-foreground)]">{field.help}</span>}
                    </label>
                  );
                }

                if (field.kind === "textarea") {
                  return (
                    <label key={field.key} className="block sm:col-span-2">
                      <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                      <textarea
                        rows={3}
                        value={String(value)}
                        placeholder={field.placeholder}
                        onChange={(event) => onChange(field.key, event.target.value)}
                        className="w-full resize-y rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs leading-5 text-[var(--foreground)] outline-none placeholder:text-[var(--muted-foreground)] focus:border-[#A9822E]"
                      />
                      {field.help && <span className="mt-1 block text-[10.5px] text-[var(--muted-foreground)]">{field.help}</span>}
                    </label>
                  );
                }

                return (
                  <label key={field.key} className="block">
                    <span className="mb-1 block text-[11.5px] font-medium text-[var(--foreground)]">{field.label}</span>
                    <input
                      value={String(value)}
                      placeholder={field.placeholder}
                      onChange={(event) => onChange(field.key, event.target.value)}
                      className="w-full rounded-lg border border-[var(--border)] bg-[var(--background)] px-3 py-2 text-xs text-[var(--foreground)] outline-none placeholder:text-[var(--muted-foreground)] focus:border-[#A9822E]"
                    />
                    {field.help && <span className="mt-1 block text-[10.5px] text-[var(--muted-foreground)]">{field.help}</span>}
                  </label>
                );
              })}
            </div>
          </section>
        ))}
      </div>
    </div>
  );
}

  const activeSpace = SPACES.find((space) => space.id === spaceId) || SPACES[0];
  const ActiveModeIcon = activeMode.icon;
  const ActiveSpaceIcon = activeSpace.icon;
  const activeOverrides = spaceConfigOverrides[spaceId];
  const activeConfiguration = buildConfiguration(spaceId, activeOverrides);

  useEffect(() => {
    let alive = true;
    Promise.all([
      fetch("/api/auth/status", { cache: "no-store" }).then(async (response) =>
        response.ok ? ((await response.json()) as AuthStatusResponse) : null,
      ),
      fetch("/api/murikah/guest-status", { cache: "no-store" }).then(async (response) =>
        response.ok ? ((await response.json()) as GuestStatusResponse) : null,
      ),
    ])
      .then(([auth, guest]) => {
        if (!alive) return;
        if (auth?.authenticated) {
          router.replace("/chat");
          return;
        }
        if (guest) {
          setRemaining(Math.max(0, Number(guest.remaining ?? 7)));
          setAuthNeeded(Boolean(guest.requires_auth));
        }
        setChecking(false);
      })
      .catch(() => {
        if (alive) setChecking(false);
      });
    return () => {
      alive = false;
    };
  }, [router]);

  useEffect(() => {
    endRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
  }, [messages, busy, authNeeded]);

  function updateSpaceConfiguration(key: string, value: ConfigValue) {
    setSpaceConfigOverrides((current) => ({
      ...current,
      [spaceId]: {
        ...current[spaceId],
        [key]: value,
      },
    }));
  }

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const prompt = draft.trim();
    if (!prompt || busy) return;
    if (remaining <= 0 || authNeeded) {
      setAuthNeeded(true);
      return;
    }

    const prior = messages.slice(-6);
    setMessages((current) => [...current, { role: "user", content: prompt }]);
    setDraft("");
    setBusy(true);
    setError("");

    try {
      const response = await fetch("/api/murikah/guest-chat", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          prompt,
          history: prior,
          mode: activeMode.id,
          space: activeSpace.id,
          configuration: activeConfiguration,
        }),
      });
      const data = (await response.json().catch(() => ({}))) as GuestChatResponse;
      if (!response.ok) {
        if (response.status === 403) {
          setRemaining(0);
          setAuthNeeded(true);
          return;
        }
        throw new Error(
          typeof data.detail === "string" ? data.detail : "Tutor could not answer that prompt.",
        );
      }
      setMessages((current) => [
        ...current,
        { role: "assistant", content: String(data.answer || "") },
      ]);
      const nextRemaining = Math.max(0, Number(data.remaining) || 0);
      setRemaining(nextRemaining);
      setAuthNeeded(Boolean(data.requires_auth) || nextRemaining === 0);
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "Tutor could not answer that prompt.");
    } finally {
      setBusy(false);
    }
  }

  if (checking) {
    return (
      <main className="grid min-h-screen place-items-center bg-[var(--background)] px-6 text-sm text-[var(--muted-foreground)]">
        Opening Murikah Tutor…
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-[var(--background)] text-[var(--foreground)]">
      <div className="flex min-h-screen">
        <aside className="sticky top-0 hidden h-screen w-64 shrink-0 flex-col border-r border-[var(--border)]/70 bg-[var(--background)] px-3 py-4 lg:flex">
          <div className="mb-5 px-2">
            <div className="inline-flex items-center gap-2.5 rounded-lg bg-[#1E2A30] px-3 py-1.5 text-sm font-semibold tracking-tight text-white shadow-sm">
              <span>Murikah</span>
              <span aria-hidden className="h-4 w-px bg-[#A9822E]" />
              <span>Tutor</span>
            </div>
          </div>
          <nav className="space-y-1" aria-label="Learning spaces">
            {SPACES.map((option) => (
              <SidebarButton
                key={option.id}
                option={option}
                active={option.id === spaceId}
                onSelect={() => setSpaceId(option.id)}
              />
            ))}
          </nav>
          <div className="mt-auto border-t border-[var(--border)]/70 px-2 pt-4">
            <Link
              href="/login?next=/chat"
              className="block rounded-lg px-3 py-2 text-sm font-medium text-[var(--foreground)]/75 hover:bg-[var(--muted)] hover:text-[var(--foreground)]"
            >
              Sign in
            </Link>
            <p className="mt-2 px-3 text-[11px] text-[var(--muted-foreground)]">Apache-2.0</p>
          </div>
        </aside>

        <div className="min-w-0 flex-1">
          <header className="sticky top-0 z-30 border-b border-[var(--border)]/70 bg-[var(--background)]/95 backdrop-blur">
            <div className="flex h-16 items-center justify-between px-4 sm:px-6">
              <div className="inline-flex items-center gap-2.5 rounded-lg bg-[#1E2A30] px-3 py-1.5 text-sm font-semibold tracking-tight text-white shadow-sm lg:hidden">
                <span>Murikah</span>
                <span aria-hidden className="h-4 w-px bg-[#A9822E]" />
                <span>Tutor</span>
              </div>
              <div className="hidden text-sm font-medium text-[var(--muted-foreground)] lg:block">
                {activeSpace.label}
              </div>
              <Link
                href="/login?next=/chat"
                className="text-sm font-medium text-[var(--foreground)]/75 hover:text-[var(--foreground)]"
              >
                Sign in
              </Link>
            </div>
          </header>

          <div className="border-b border-[var(--border)]/60 px-4 py-2 lg:hidden">
            <div className="flex gap-2 overflow-x-auto pb-1" aria-label="Learning spaces">
              {SPACES.map((option) => {
                const Icon = option.icon;
                const active = option.id === spaceId;
                return (
                  <button
                    key={option.id}
                    type="button"
                    onClick={() => setSpaceId(option.id)}
                    title={option.description}
                    className={`inline-flex shrink-0 items-center gap-2 rounded-full border px-3 py-1.5 text-xs font-medium ${
                      active
                        ? "border-[#A9822E]/40 bg-[#A9822E]/10 text-[var(--foreground)]"
                        : "border-[var(--border)] text-[var(--muted-foreground)]"
                    }`}
                  >
                    <Icon size={14} />
                    {option.label}
                  </button>
                );
              })}
            </div>
          </div>

          <section className="mx-auto flex min-h-[calc(100vh-4rem)] max-w-5xl flex-col px-4 py-6 sm:px-6 sm:py-8">
            {messages.length === 0 && spaceId === "home" ? (
              <div className="mx-auto flex w-full max-w-3xl flex-1 flex-col justify-center pb-8 text-center">
                <p className="mb-3 text-xs font-semibold uppercase tracking-[0.18em] text-[#A9822E]">
                  One tutor. Every way you learn.
                </p>
                <h1 className="text-3xl font-semibold tracking-[-0.035em] sm:text-5xl">
                  Learn in the mode that fits the moment.
                </h1>
                <p className="mx-auto mt-4 max-w-2xl text-[15px] leading-7 text-[var(--muted-foreground)] sm:text-base">
                  Chat, ask, quiz, research, visualize, solve, study, read and watch without losing the thread. Explore the learning spaces in the sidebar and configure them before you sign in.
                </p>
                <div className="mx-auto mt-5 flex max-w-2xl flex-wrap justify-center gap-2 text-xs text-[var(--muted-foreground)]">
                  {["10 learning modes", "Configurable spaces", "Knowledge + memory", "Agents + partners"].map((item) => (
                    <span key={item} className="rounded-full border border-[var(--border)] bg-[var(--card)] px-3 py-1.5">
                      {item}
                    </span>
                  ))}
                </div>
              </div>
            ) : messages.length === 0 ? (
              <div className="mx-auto w-full max-w-3xl pb-4 text-left">
                <p className="text-xs font-semibold uppercase tracking-[0.16em] text-[#A9822E]">Try {activeSpace.label}</p>
                <h1 className="mt-2 text-2xl font-semibold tracking-[-0.025em] sm:text-3xl">Set it up the way you would after sign-in.</h1>
                <p className="mt-2 max-w-2xl text-sm leading-6 text-[var(--muted-foreground)]">{activeSpace.description}</p>
              </div>
            ) : (
              <div className="mx-auto w-full max-w-3xl flex-1 space-y-6 pb-8">
                {messages.map((message, index) => (
                  <article
                    key={`${message.role}-${index}`}
                    className={
                      message.role === "user"
                        ? "ml-auto max-w-[85%] rounded-2xl bg-[var(--muted)] px-4 py-3"
                        : "max-w-full"
                    }
                  >
                    {message.role === "assistant" && (
                      <div className="mb-2 text-xs font-semibold text-[#A9822E]">
                        Murikah Tutor · {activeMode.label}
                      </div>
                    )}
                    <div className="whitespace-pre-wrap text-[15px] leading-7">{message.content}</div>
                  </article>
                ))}
                {busy && <div className="text-sm text-[var(--muted-foreground)]">Murikah Tutor is thinking…</div>}
                <div ref={endRef} />
              </div>
            )}

            {authNeeded ? (
              <div className="mx-auto w-full max-w-md rounded-2xl border border-[var(--border)] bg-[var(--card)] p-6 shadow-sm">
                <div className="text-center">
                  <h2 className="text-xl font-semibold tracking-tight">Create an account to keep learning</h2>
                  <p className="mt-2 text-sm leading-6 text-[var(--muted-foreground)]">
                    You have reached the guest limit. Create an account or sign in to continue, save your progress, and keep your configured knowledge, memory, tools, agents and partners.
                  </p>
                </div>
                <div className="mt-5">
                  <MurikahSocialButtons next="/chat" />
                </div>
                <div className="my-4 flex items-center gap-3 text-xs text-[var(--muted-foreground)] before:h-px before:flex-1 before:bg-[var(--border)] after:h-px after:flex-1 after:bg-[var(--border)]">
                  or
                </div>
                <Link
                  href="/login?next=/chat"
                  className="block w-full rounded-lg bg-[#1E2A30] px-4 py-2.5 text-center text-sm font-semibold text-white hover:opacity-90"
                >
                  Sign in with username
                </Link>
                <p className="mt-4 text-center text-xs text-[var(--muted-foreground)]">Apache-2.0</p>
              </div>
            ) : (
              <form onSubmit={submit} className="mx-auto w-full max-w-3xl">
                <SpaceConfigurator
                  space={activeSpace}
                  values={activeOverrides}
                  onChange={updateSpaceConfiguration}
                />

                <div className="mb-3 flex items-start gap-3 rounded-xl border border-[var(--border)]/80 bg-[var(--card)] px-3.5 py-3 text-left shadow-sm">
                  <span className="mt-0.5 rounded-lg bg-[#A9822E]/10 p-2 text-[#A9822E]">
                    <ActiveSpaceIcon size={16} />
                  </span>
                  <div className="min-w-0 flex-1">
                    <div className="text-xs font-semibold text-[var(--foreground)]">{activeSpace.label}</div>
                    <p className="mt-0.5 text-xs leading-5 text-[var(--muted-foreground)]">{activeSpace.description}</p>
                  </div>
                </div>

                {error && (
                  <p className="mb-2 rounded-lg bg-red-500/10 px-3 py-2 text-sm text-red-600">{error}</p>
                )}
                <div className="rounded-2xl border border-[var(--border)] bg-[var(--card)] p-2 shadow-sm focus-within:ring-2 focus-within:ring-[#A9822E]/30">
                  <textarea
                    value={draft}
                    onChange={(event) => setDraft(event.target.value)}
                    onKeyDown={(event) => {
                      if (event.key === "Enter" && !event.shiftKey) {
                        event.preventDefault();
                        event.currentTarget.form?.requestSubmit();
                      }
                    }}
                    placeholder={activeMode.placeholder}
                    rows={3}
                    maxLength={6000}
                    className="w-full resize-none bg-transparent px-3 py-2 text-[15px] outline-none placeholder:text-[var(--muted-foreground)]"
                  />

                  <div className="flex items-end justify-between gap-3 px-2 pb-1">
                    <div className="relative">
                      <button
                        type="button"
                        onClick={() => setModeMenuOpen((open) => !open)}
                        aria-haspopup="menu"
                        aria-expanded={modeMenuOpen}
                        className="inline-flex items-center gap-2 rounded-lg px-2 py-1.5 text-sm font-semibold text-[var(--foreground)] hover:bg-[var(--muted)]"
                      >
                        <ActiveModeIcon size={17} className="text-[#A9822E]" />
                        <span>{activeMode.label}</span>
                        <ChevronDown size={15} className="text-[var(--muted-foreground)]" />
                      </button>

                      {modeMenuOpen && (
                        <div
                          role="menu"
                          className="absolute bottom-[calc(100%+10px)] left-0 z-50 max-h-[420px] w-[min(430px,calc(100vw-3rem))] overflow-y-auto rounded-2xl border border-[var(--border)] bg-[var(--card)] p-2 shadow-xl"
                        >
                          <div className="px-3 pb-2 pt-1 text-[11px] font-semibold uppercase tracking-[0.14em] text-[var(--muted-foreground)]">
                            Choose how you want to learn
                          </div>
                          {MODES.map((mode) => {
                            const Icon = mode.icon;
                            const active = mode.id === modeId;
                            return (
                              <button
                                key={mode.id}
                                type="button"
                                role="menuitem"
                                onClick={() => {
                                  setModeId(mode.id);
                                  setModeMenuOpen(false);
                                }}
                                className={`flex w-full items-start gap-3 rounded-xl px-3 py-2.5 text-left transition-colors ${
                                  active ? "bg-[var(--muted)]" : "hover:bg-[var(--muted)]/70"
                                }`}
                              >
                                <Icon size={18} className="mt-0.5 shrink-0 text-[#A9822E]" />
                                <span className="min-w-0 flex-1">
                                  <span className="flex items-center gap-2 text-sm font-semibold text-[var(--foreground)]">
                                    {mode.label}
                                    {active && <Sparkles size={13} className="text-[#A9822E]" />}
                                  </span>
                                  <span className="mt-0.5 block text-xs leading-5 text-[var(--muted-foreground)]">
                                    {mode.description}
                                  </span>
                                </span>
                              </button>
                            );
                          })}
                        </div>
                      )}
                    </div>

                    <button
                      type="submit"
                      disabled={busy || !draft.trim()}
                      className="rounded-lg bg-[#1E2A30] px-4 py-2 text-sm font-semibold text-white disabled:cursor-not-allowed disabled:opacity-40"
                    >
                      Send
                    </button>
                  </div>
                </div>
                <p className="mt-3 text-center text-[11px] text-[var(--muted-foreground)]">Apache-2.0</p>
              </form>
            )}
          </section>
        </div>
      </div>
    </main>
  );
}
