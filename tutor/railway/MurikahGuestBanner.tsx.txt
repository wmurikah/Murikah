"use client";
import { useEffect, useState } from "react";
import { usePathname } from "next/navigation";

type Status = { guest: boolean; remaining?: number; requires_auth?: boolean };
export default function MurikahGuestBanner() {
  const path = usePathname();
  const [status, setStatus] = useState<Status | null>(null);
  useEffect(() => {
    let alive = true;
    async function refresh() {
      try {
        const res = await fetch("/api/murikah/access/status", { cache: "no-store" });
        if (res.ok) { const value = await res.json(); if (alive) setStatus(value); }
      } catch { /* Keep the last confirmed allowance during a network interruption. */ }
    }
    void refresh();
    const timer = window.setInterval(refresh, 2000);
    window.addEventListener("focus", refresh);
    return () => { alive = false; clearInterval(timer); window.removeEventListener("focus", refresh); };
  }, []);
  if (!status?.guest || !status.requires_auth) return null;
  const next = encodeURIComponent(path || "/chat");
  const preserveWork = () => window.dispatchEvent(new Event("murikah:account-navigation"));
  return <div role="alert" className="flex shrink-0 flex-wrap items-center justify-between gap-2 border-b border-[#A9822E]/25 bg-[#A9822E]/10 px-5 py-3 text-sm">
    <div><p className="font-semibold">Continue your work</p><p>You've reached the free guest limit. Sign up to keep your guest workspace and continue, or sign in to your account.</p></div>
    <div className="flex items-center gap-4">
      <a onClick={preserveWork} href={`/login?next=${next}`} className="font-medium">Sign in</a>
      <a onClick={preserveWork} href={`/register?next=${next}`} className="rounded-lg bg-[#1E2A30] px-4 py-2 font-semibold text-white">Sign up</a>
    </div>
  </div>;
}
