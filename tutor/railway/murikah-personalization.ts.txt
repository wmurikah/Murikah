export const COMPOSER_PLACEHOLDERS = [
  "What are you working on?",

  "What do you want to explore?",
  "Where should we start?",
  "What's the question?",
  "Tell me what you're working through.",

function bucketForHour(hour: number): GreetingTemplate["timeBucket"] {
  if (hour >= 5 && hour < 12) return "morning";
  if (hour >= 12 && hour < 17) return "afternoon";
  if (hour >= 17 && hour < 22) return "evening";
  return "late";
}

export function greetingTemplatesForContext(
  context: GreetingContext,
): GreetingTemplate[] {
  const bucket = bucketForHour(context.localHour);
  return GREETINGS.filter(
    (template) =>
      (template.category !== "time" || template.timeBucket === bucket) &&
      (template.category !== "returning" || context.returningUser !== false),
  );
}

export function greetingForContext(
  context: GreetingContext,
  choiceIndex: number,
): string {
  const pool = greetingTemplatesForContext(context);
  const template = pool[Math.abs(Math.trunc(choiceIndex)) % pool.length];
  const name = String(context.name || "").trim();
  return (name ? template.named.replaceAll("{name}", name) : template.generic).trim();
}

export function composerPlaceholderForIndex(choiceIndex: number): string {
  return COMPOSER_PLACEHOLDERS[
    Math.abs(Math.trunc(choiceIndex)) % COMPOSER_PLACEHOLDERS.length
  ];
}

const ENTITY_RE = /&mdash;|&#8212;|&#x0*2014;/gi;
const CODE_RE = /```[sS]*?```|`[^`
]*`/g;
const URL_RE = /https?://[^s<>()]+/gi;

function sanitizeCode(value: string): string {
  return value.replaceAll("—", "\u2014").replace(ENTITY_RE, "\u2014");
}

function sanitizeUrl(value: string): string {
  return value
    .replaceAll("—", "%E2%80%94")
    .replace(/&mdash;/gi, "%26mdash%3B")
    .replace(/&#8212;/gi, "%26%238212%3B")
    .replace(/&#x0*2014;/gi, "%26%23x2014%3B");
}

function sanitizePlain(value: string): string {
  return value
    .replaceAll(" — ", "; ")
    .replaceAll("—", ", ")
    .replace(ENTITY_RE, "; ");
}

function sanitizePlainWithUrls(value: string): string {
  let output = "";
  let cursor = 0;
  for (const match of value.matchAll(URL_RE)) {
    const start = match.index ?? 0;
    output += sanitizePlain(value.slice(cursor, start));
    output += sanitizeUrl(match[0]);
    cursor = start + match[0].length;
  }
  return output + sanitizePlain(value.slice(cursor));
}

export function sanitizeTutorVisibleText(value: string): string {
  let output = "";
  let cursor = 0;
  for (const match of value.matchAll(CODE_RE)) {
    const start = match.index ?? 0;
    output += sanitizePlainWithUrls(value.slice(cursor, start));
    output += sanitizeCode(match[0]);
    cursor = start + match[0].length;
  }
  return output + sanitizePlainWithUrls(value.slice(cursor));
}
