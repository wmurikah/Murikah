function response(body: unknown, status = 200) {
  return Promise.resolve(new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json" } }));
}

function installFetch(workspaceState: "guest" | "none" = "none") {
  return vi.spyOn(globalThis, "fetch").mockImplementation((input, init) => {
    const url = String(input);
    if (url.endsWith("/workspace")) return response({ state: workspaceState, simulation: true, sections: [] });
    if (url.includes("/catalog/facets")) return response(facets);
    if (url.includes("/catalog/internal-audit-intern")) return response({ ok: true, internship: detail });
    if (url.includes("/catalog?") || url.endsWith("/catalog")) return response({ ok: true, count: 2, results: [qualifying, practice] });
    if (url.endsWith("/start") && init?.method === "POST") {
      return response({ state: "active", simulation: true, sections: [], internship: { internship_id: "vi_started" }, company: {}, tasks: [], people: [] });
    }
    return response({ error: "not_found" }, 404);
  });
}

describe("Virtual Internship Phase 10 catalog", () => {
  beforeEach(() => {
    nav.pathname = "/virtual-internship";
    nav.search = "";
    nav.push.mockReset();
    nav.replace.mockReset();
    nav.refresh.mockReset();
  });

  afterEach(() => {
    cleanup();
    vi.restoreAllMocks();
  });

  it("shows the marketplace for a member with no active internship", async () => {
    installFetch("none");
    render(<MurikahVirtualInternshipWorkspace />);

    expect(await screen.findByRole("heading", { level: 1, name: "Choose your Virtual Internship" })).toBeTruthy();
    expect(screen.getByRole("searchbox", { name: "Search internships" })).toBeTruthy();
    expect(screen.getByLabelText("Career family")).toBeTruthy();
    expect(screen.getByLabelText("Sector")).toBeTruthy();
    expect(screen.getByLabelText("Type")).toBeTruthy();
    expect(screen.getByLabelText("Weekly workload")).toBeTruthy();
    expect(screen.getByLabelText("Experience")).toBeTruthy();
    expect(screen.getByText("Qualifying Virtual Internships")).toBeTruthy();
    expect(screen.getByText("Practice internships")).toBeTruthy();
    expect(screen.getByText("Aster Vale Services Group")).toBeTruthy();
    expect(screen.getByText("Minimum 90 calendar days")).toBeTruthy();
    expect(screen.getByText("8-12 hours/week")).toBeTruthy();
    expect(screen.getByText("Does not qualify for completion credentials")).toBeTruthy();
    expect(screen.getAllByRole("heading", { level: 1 })).toHaveLength(1);
  });

  it("lets guests browse but routes a qualifying start through existing sign in", async () => {
    installFetch("guest");
    nav.pathname = "/virtual-internship/catalog/internal-audit-intern";
    nav.search = "version=sv_internal_audit_qualifying_v1";
    render(<MurikahVirtualInternshipWorkspace />);

    expect(await screen.findByRole("heading", { level: 1, name: "Internal Audit Intern" })).toBeTruthy();
    const link = screen.getByRole("link", { name: "Sign in to review commitment" });
    expect(link.getAttribute("href")).toContain("/login?next=");
    expect(link.getAttribute("href")).toContain("commitment");
  });

  it("preserves search and filter choices in the catalog URL", async () => {
    installFetch("none");
    render(<MurikahVirtualInternshipWorkspace />);
    await screen.findByRole("heading", { level: 1, name: "Choose your Virtual Internship" });

    fireEvent.change(screen.getByRole("searchbox", { name: "Search internships" }), { target: { value: "audit" } });
    fireEvent.click(screen.getByRole("button", { name: "Search" }));
    expect(nav.replace).toHaveBeenCalledWith("/virtual-internship?q=audit");

    fireEvent.change(screen.getByLabelText("Career family"), { target: { value: "audit_assurance" } });
    expect(nav.replace).toHaveBeenCalledWith("/virtual-internship?career=audit_assurance");
  });

  it("requires commitment acknowledgements and pins the reviewed scenario version", async () => {
    const fetchMock = installFetch("none");
    nav.pathname = "/virtual-internship/catalog/internal-audit-intern/commitment";
    nav.search = "version=sv_internal_audit_qualifying_v1";
    render(<MurikahVirtualInternshipWorkspace />);

    const start = await screen.findByRole("button", { name: "Start internship" });
    expect((start as HTMLButtonElement).disabled).toBe(true);
    const checks = screen.getAllByRole("checkbox");
    expect(checks).toHaveLength(4);
    checks.forEach((box) => fireEvent.click(box));
    expect((start as HTMLButtonElement).disabled).toBe(false);
    fireEvent.click(start);

    await waitFor(() => {
      const call = fetchMock.mock.calls.find(([url, init]) => String(url).endsWith("/start") && init?.method === "POST");
      expect(call).toBeTruthy();
      const payload = JSON.parse(String(call?.[1]?.body || "{}"));
      expect(payload.scenario_slug).toBe("internal-audit-intern");
      expect(payload.scenario_version_id).toBe("sv_internal_audit_qualifying_v1");
      expect(payload.qualifying).toBeUndefined();
      expect(payload.mode).toBeUndefined();
    });
  });

  it("shows a distinct no-results state for filters instead of the old internship empty state", async () => {
    vi.spyOn(globalThis, "fetch").mockImplementation((input) => {
      const url = String(input);
      if (url.endsWith("/workspace")) return response({ state: "none", simulation: true, sections: [] });
      if (url.includes("/catalog/facets")) return response(facets);
      if (url.includes("/catalog?")) return response({ ok: true, count: 0, results: [] });
      return response({ error: "not_found" }, 404);
    });
    nav.search = "q=nomatch";
    render(<MurikahVirtualInternshipWorkspace />);

    expect(await screen.findByText("No internships match those filters.")).toBeTruthy();
    expect(screen.queryByText("No internship is available right now.")).toBeNull();
    expect(screen.getByRole("button", { name: "Clear filters" })).toBeTruthy();
  });
});
