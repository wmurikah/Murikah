function jsonResponse(value: unknown, status = 200) {
  return new Response(JSON.stringify(value), {
    status,
    headers: { "content-type": "application/json" },
  });
}

function workspaceFetch(extra?: (url: string, init?: RequestInit) => Response | Promise<Response> | undefined) {
  return vi.fn(async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = String(input);
    const custom = extra?.(url, init);
    if (custom) return await custom;
    if (url.endsWith("/workspace")) return jsonResponse(workspace);
    throw new Error("Unexpected request: " + url);
  });
}

    nav.pathname = "/virtual-internship/documents";
    const fetchMock = workspaceFetch((url) => {
      if (url.includes("/documents/fact%3Apolicy_1")) {
        return jsonResponse({
          ok: true,
          document: {
            ...workspace.documents[0],
            content: "Monthly cycle counts are required before close.",
          },
        });
      }
      return undefined;
    });
    vi.stubGlobal("fetch", fetchMock);
    render(<MurikahVirtualInternshipWorkspace />);

    fireEvent.click(await screen.findByRole("button", { name: "Open document" }));
    expect(await screen.findByText("Monthly cycle counts are required before close.")).toBeVisible();
    expect(fetchMock).toHaveBeenCalledWith(
      expect.stringContaining("/documents/fact%3Apolicy_1?internship_id=vi_1"),
      expect.objectContaining({ credentials: "include" }),
    );
    expect(screen.queryByRole("button", { name: /upload/i })).toBeNull();
  });

  it("streams a workplace actor reply through the Phase 4 API surface", async () => {
    nav.pathname = "/virtual-internship/inbox";
    const fetchMock = workspaceFetch((url, init) => {
      if (url.endsWith("/actor/messages/stream")) {
        expect(init?.method).toBe("POST");
        const body = JSON.parse(String(init?.body));
        expect(body.scenario_actor_id).toBe("actor_supervisor");
        expect(body.request_id).toMatch(/^actor-/);
        return new Response(
          JSON.stringify({ type: "chunk", text: "Please start with the policy and variance evidence." }) + "\n" +
          JSON.stringify({ type: "final", text: "Please start with the policy and variance evidence.", message_id: "vim_2" }) + "\n",
          { headers: { "content-type": "application/x-ndjson" } },
        );
      }
      return undefined;
    });
    vi.stubGlobal("fetch", fetchMock);
    render(<MurikahVirtualInternshipWorkspace />);

    fireEvent.click(await screen.findByRole("button", { name: /Lina A\./ }));
    const composer = screen.getByLabelText("Workplace message");
    fireEvent.change(composer, { target: { value: "Where should I start?" } });
    fireEvent.click(screen.getByRole("button", { name: "Send" }));

    expect(await screen.findByText("Please start with the policy and variance evidence.")).toBeVisible();
    expect(fetchMock).toHaveBeenCalledWith(
      expect.stringContaining("/actor/messages/stream"),
      expect.objectContaining({ method: "POST" }),
    );
  });

    nav.pathname = "/virtual-internship/mentor";
